     1→// ============================================================
     2→// Obscura — 本项目自研轻量无头浏览器引擎（--stealth 隐身模式）
     3→//
     4→// 设计要点:
     5→//  - 基于 Playwright chromium 复用: 全局单例浏览器常驻 + 页面池(默认并发 2),
     6→//    避免每次渲染冷启动 launch(~1s+) 的开销
     7→//  - 默认开启 --stealth 隐身模式: STEALTH_INIT_SCRIPTS(静态抹平)+buildIdentityInitScript(按 UA 参数化身份脚本)
     8→//    在每个页面文档创建前注入(全 frame), 抹平 navigator.webdriver / window.chrome / plugins /
     9→//    WebGL / canvas / userAgentData / permissions 等 常见自动化指纹 (参考 puppeteer-extra-stealth
    10→//    思路的本地精简实现); hh-d2: 另经 CDP Network.setUserAgentOverride(+userAgentMetadata) 让
    11→//    【网络层 sec-ch-ua* 头】与 JS userAgentData 与 UA 字符串三方自洽(头组侧 JS 抹平不可达)
    12→//  - 指纹随机化: 每个域名槽位首次创建时随机 视口/UA/dpr/locale/时区(移动性按 UA 派生, hh-d2)
    13→//  - 挑战自动等待: 命中 Cloudflare("Just a moment...")/JS跳转壳等特征时,
    14→//    每 1s 轮询 content 最多 challengeWaitMs, 等待站点侧自动放行
    15→//  - Cookie 回传: 渲染完成后将 context.cookies() 转成 Set-Cookie 风格字符串
    16→//    返回给 fetcher 写入 CookieJar —— HTTP 引擎与浏览器引擎凭证打通的关键
    17→//    (如 cf_clearance 回流后, 纯 HTTP 抓取也能过盾)
    18→//
    19→// 隔离说明(重要): 页面池的每个槽位持有独立 BrowserContext(浏览器级隔离)。
    20→//  - 同域名复用同一 context: 保留 cookie(挑战凭证/会话), 指纹稳定, 像真实回访用户
    21→//  - 跨域名复用前销毁重建 context 并换新指纹: 杜绝 cookie/localStorage 跨站串扰
    22→//    (若未来出现同域多账号等更细粒度隔离需求, 可将槽位桶改为 Map<domain, Slot[]>)
    23→// ============================================================

   102→function pick<T>(arr: T[]): T {
   103→  return arr[Math.floor(Math.random() * arr.length)]
   104→}
   105→function jitter(n: number, max: number): number {
   106→  const d = Math.floor(Math.random() * (max * 2 + 1)) - max
   107→  return Math.max(1, n + d)
   108→}

   110→/** UA 移动性判定(与 fetcher.isMobileUa 同语义; obscura 禁止反向 import fetcher 防循环依赖, 本地复制) */
   111→export function isMobileUaLocal(ua: string): boolean {
   112→  return /iPhone|iPad|Android|Mobile Safari|;\s*Mobile\//.test(ua)

   115→/** 随机生成一份浏览器指纹(桌面/移动池 + 小幅抖动)
   116→ *  hh-d2: 传入 UA 覆盖时移动性按 UA 派生(原先纯随机 22% —— Android UA 可能拿到桌面视口/
   117→ *  无触屏 context, sec-ch-ua-mobile ?1 配 maxTouchPoints=0 自相矛盾, 探针实证暴露)
   118→ *  E3: locale/timezone/deviceScaleFactor 加权池 —— 原实现 locale/timezone 恒 zh-CN/Asia/Shanghai,
   119→ *  deviceScaleFactor 仅桌面 1/2 二值+移动 1/2 二值, 与真机分布偏离(creepjs 类检测会聚成同一指纹面);
   120→ *  引入 zh-TW/en-US/en-GB 少量配比(总权重 ≤30%, 主流仍中文), deviceScaleFactor 加入 1.25/1.5
   121→ *  高分屏常见值, 桌面 1 仍占多数(60%), 移动以 2/3 为主, 1.5 次之 */
   122→// E3: locale/timezone 加权配对池 —— 与 Accept-Language 头组(newStealthContext 动态构造)及
   123→// 静态脚本 4(per-context 动态 init 覆盖)三方对齐
   124→const LOCALE_POOL: Array<{ locale: string; timezone: string; weight: number }> = [
   125→  { locale: 'zh-CN', timezone: 'Asia/Shanghai', weight: 70 },
   126→  { locale: 'zh-TW', timezone: 'Asia/Taipei', weight: 12 },
   127→  { locale: 'en-US', timezone: 'America/New_York', weight: 12 },
   128→  { locale: 'en-GB', timezone: 'Europe/London', weight: 6 },
   129→]
   130→function pickLocale(): { locale: string; timezone: string } {
   131→  const total = LOCALE_POOL.reduce((s, e) => s + e.weight, 0)
   132→  let r = Math.random() * total
   133→  for (const e of LOCALE_POOL) {
   134→    if (r < e.weight) return { locale: e.locale, timezone: e.timezone }
   135→    r -= e.weight
   136→  }
   137→  return { locale: 'zh-CN', timezone: 'Asia/Shanghai' }
   138→}
   139→// E3: deviceScaleFactor 桌面加权池 —— 1 占多数(普通显示器), 1.25/1.5 高分屏常见, 2 Retina
   140→function pickDesktopDsf(): number {
   141→  const r = Math.random()
   142→  if (r < 0.6) return 1
   143→  if (r < 0.7) return 1.25
   144→  if (r < 0.85) return 1.5
   145→  return 2
   146→}
   147→// E3: deviceScaleFactor 移动加权池 —— 2/3 为主流手机, 1.5 次之, 1 少量低端机
   148→function pickMobileDsf(): number {
   149→  const r = Math.random()
   150→  if (r < 0.1) return 1
   151→  if (r < 0.25) return 1.5
   152→  if (r < 0.85) return 2
   153→  return 3
   154→}

   156→export function randomFingerprint(overrides?: { userAgent?: string }): ObscuraFingerprint {
   157→  const mobile = overrides?.userAgent ? isMobileUaLocal(overrides.userAgent) : Math.random() < 0.22
   158→  const { locale, timezone } = pickLocale()
   159→  if (mobile) {
   160→    const [w, h] = pick(MOBILE_SIZES)
   161→    return {
   162→      viewport: { width: w, height: h },
   163→      userAgent: overrides?.userAgent || pick(MOBILE_UAS),
   164→      locale,
   165→      timezoneId: timezone,
   166→      colorScheme: 'light',
   167→      deviceScaleFactor: pickMobileDsf(),
   168→      mobile: true,
   169→    }
   170→  }
   171→  const [w, h] = pick(DESKTOP_SIZES)
   172→  return {
   173→    viewport: { width: jitter(w, 8), height: jitter(h, 8) },
   174→    userAgent: overrides?.userAgent || pick(DESKTOP_UAS),
   175→    locale,
   176→    timezoneId: timezone,
   177→    colorScheme: 'light',
   178→    deviceScaleFactor: pickDesktopDsf(),
   179→    mobile: false,
   180→  }
   181→}

   183→/** E3: 按 fp.locale 派生 Accept-Language 头 —— 与 newContext(locale)(影响 navigator.language 与
   184→ *  默认 Accept-Language) 及 per-context 动态 init 脚本(覆盖 navigator.language/languages)三方对齐 */
   185→function acceptLanguageFor(locale: string): string {
   186→  if (locale.startsWith('zh-CN')) return 'zh-CN,zh;q=0.9,en;q=0.6'
   187→  if (locale.startsWith('zh-TW')) return 'zh-TW,zh;q=0.9,en;q=0.6'
   188→  if (locale.startsWith('en-US')) return 'en-US,en;q=0.9,zh;q=0.6'
   189→  if (locale.startsWith('en-GB')) return 'en-GB,en;q=0.9,zh;q=0.6'
   190→  return 'zh-CN,zh;q=0.9,en;q=0.6'
   191→}

   231→/** 从 UA 解析身份(brands 结构与 ff 轮 fingerprintHeadersFor 同构: Chromium/Google Chrome[/Edge]/Not:A-Brand) */
   232→export function parseUaIdentity(ua: string): UaIdentity {
   233→  const family: UaFamily = /Firefox\//.test(ua) ? 'firefox' : /Chrome\/|\bEdg\b/.test(ua) ? 'chromium' : /Safari\//.test(ua) ? 'safari' : 'unknown'
   234→  const os: UaIdentity['os'] = /Android/.test(ua) ? 'android' : /iPhone|iPad|iPod/.test(ua) ? 'ios' : /Windows/.test(ua) ? 'windows' : /Mac OS|Macintosh/.test(ua) ? 'macos' : /X11|Linux|CrOS/.test(ua) ? 'linux' : 'other'
   235→  const mobile = isMobileUaLocal(ua)
   236→  const chromeVer = ua.match(/Chrome\/(\d+)/)?.[1] || ''
   237→  const edgeVer = ua.match(/Edg(?:e|A|iOS)?\/(\d+)/)?.[1] || ''
   238→  const navPlatform = /Android/.test(ua) ? 'Linux armv8l' : /iPad/.test(ua) ? 'MacIntel' : /iPhone/.test(ua) ? 'iPhone' : /Windows/.test(ua) ? 'Win32' : /Mac OS|Macintosh/.test(ua) ? 'MacIntel' : 'Linux x86_64'
   239→  const chPlatform = /Android/.test(ua) ? 'Android' : /iPhone|iPad|iPod/.test(ua) ? 'iOS' : /Windows/.test(ua) ? 'Windows' : /Mac OS|Macintosh/.test(ua) ? 'macOS' : /CrOS/.test(ua) ? 'Chrome OS' : /X11|Linux/.test(ua) ? 'Linux' : 'Windows'
   240→  // ii-c 修前缺陷: 旧正则 /Android[^;)]*[;)]\s*([^;)]+?)(?:\s+Build\/\S+)?\s*\)/ 只认
   241→  // "Android <ver>; <model>"两段形态, 三段 UA(Linux; U; Android 13; zh-cn; M2102J2SC Build/…)
   242→  // 在捕获前存在 locale 中间段 → 整体失配 model=''(探针实证), 与 UA 字符串内可见机型矛盾;
   243→  // 改为允许任意个 "; 分隔中间段后取最后一个段为机型(每段不含 ;/), 分词无歧义无回溯风险
   244→  const model = /Android/.test(ua) ? ((ua.match(/Android[^;)]*(?:;[^;)]+)*;\s*([^;)]+?)(?:\s+Build\/\S+)?\s*\)/)?.[1] || '').trim()) : ''
   245→  const platformVersion = /Android/.test(ua) ? (ua.match(/Android\s+([\d.]+)/)?.[1] || '') : /iPhone|iPad|iPod/.test(ua) ? (ua.match(/OS\s+(\d+[._]\d+)/)?.[1]?.replace('_', '.') || '') : /Windows/.test(ua) ? '10.0.0' : /Mac OS|Macintosh/.test(ua) ? '10.15.7' : ''
   246→  const fullVersion = chromeVer ? `${chromeVer}.0.0.0` : ''
   247→  const brands: Array<{ brand: string; version: string }> = []
   248→  const fullVersionList: Array<{ brand: string; version: string }> = []
   249→  if (chromeVer) {
   250→    brands.push({ brand: 'Chromium', version: chromeVer }, { brand: 'Google Chrome', version: chromeVer })
   251→    fullVersionList.push({ brand: 'Chromium', version: fullVersion }, { brand: 'Google Chrome', version: fullVersion })
   252→    if (edgeVer) {
   253→      brands.push({ brand: 'Microsoft Edge', version: edgeVer })
   254→      fullVersionList.push({ brand: 'Microsoft Edge', version: `${edgeVer}.0.0.0` })
   255→    }
   256→    brands.push({ brand: 'Not:A-Brand', version: '24' })
   257→    fullVersionList.push({ brand: 'Not:A-Brand', version: '24.0.0.0' })
   258→  }
   259→  return {
   260→    ua,
   261→    family,
   262→    os,
   263→    mobile,
   264→    chromeVer,
   265→    edgeVer,
   266→    fullVersion,
   267→    navPlatform,
   268→    chPlatform,
   269→    vendor: family === 'chromium' ? 'Google Inc.' : family === 'safari' ? 'Apple Computer, Inc.' : '',
   270→    model,
   271→    platformVersion,
   272→    architecture: os === 'android' || os === 'ios' ? 'arm' : 'x86',
   273→    bitness: '64',
   274→    brands,
   275→    fullVersionList,
   276→    gpu: GPU_BY_OS[os],
   277→  }
   278→}

   292→    bitness: id.bitness,
   293→    model: id.model,
   294→    wow64: false,
   295→  }

   298→/**
   299→ * 按 UA 参数化的身份脚本(附加在 STEALTH_INIT_SCRIPTS 之后注入, 每个 frame 文档创建前执行):
   300→ *  - UA/appVersion/platform/vendor/productSub 钉到身份值
   301→ *  - Chromium 家族: navigator.userAgentData(brands/mobile/platform + getHighEntropyValues 全量高熵)
   302→ *  - 非 Chromium(Safari/Firefox 语义): userAgentData 整体摘除(own+原型链 delete,
   303→ *    in 判定与真 Safari 同为 false; 原型槽不可删时退回 own undefined 遮蔽)+ window.chrome 删除

   699→/** JS 跳转壳: 极短 + location 跳转脚本, 或 meta refresh 短页 */
   700→export function isJsRedirectShell(html: string): boolean {
   701→  const s = (html || '').trim()
   702→  if (!s) return true
   703→  const hasRedirect = /window\.location\s*[.[]|location\.href\s*=|location\.replace\s*\(|location\.assign\s*\(/.test(s)
   704→  const hasRefresh = /http-equiv\s*=\s*["']?refresh/i.test(s)
   705→  if (s.length < 1200 && hasRedirect) return true
   706→  if (s.length < 1200 && hasRefresh) return true
   707→  return false
   708→}

   710→function hasNormalTitle(html: string): boolean {
   711→  const m = html.match(/<title[^>]*>([\s\S]{1,300}?)<\/title>/i)
   712→  if (!m) return false
   713→  const t = m[1].trim().toLowerCase()
   714→  if (t.length < 2) return false
   715→  // 标题本身就是挑战提示的不算正常标题
   716→  const bad = ['just a moment', 'attention required', 'access denied', 'forbidden', '请开启', '验证', '驗證', '请稍候', '请稍後']
   717→  return !bad.some((k) => t.includes(k))
   718→}

   720→/** 浏览器侧挑战判定: 命中强特征 / 软特征且非"长+正常标题"页 / JS跳转壳 */
   721→export function looksLikeChallenge(html: string): boolean {
   722→  if (isJsRedirectShell(html)) return true
   723→  const lower = (html || '').toLowerCase()
   724→  // CF JS Detections 脚本(challenge-platform/scripts/jsd/main.js)是 Bot Management 下
   725→  // 正常页面普遍内嵌的探测脚本 —— 页面有正常标题且足够长时不算挑战(否则真实内容页
   726→  // 会误触发 40s 挑战等待循环后抛超时, 101kks 实测)
   727→  const jsdBenign = lower.includes('challenge-platform/scripts/jsd') && lower.length >= 1200 && hasNormalTitle(html)
   728→  if (!jsdBenign && CHALLENGE_STRONG_MARKERS.some((k) => lower.includes(k))) return true
   729→  if (CHALLENGE_SOFT_MARKERS.some((k) => lower.includes(k))) {
   730→    // 长且带正常标题的真实内容页, 不因正文/导航提及"验证码"等词误判
   731→    if (lower.length >= 1200 && hasNormalTitle(html)) return false
   732→    return true
   733→  }
   734→  return false
   735→}

   845→async function ensureBrowser(): Promise<Browser> {
   846→  if (S.browser && S.browser.isConnected()) return S.browser
   847→  if (S.launchPromise) return S.launchPromise
   848→  S.launchPromise = (async () => {
   849→    if (!S.pwModule) {
   850→      // 惰性 import('playwright'): 避免未用浏览器引擎时加载 ~几十ms 的模块开销
   851→      S.pwModule = await import('playwright')
   852→    }
   853→    // playwright 类型无 any 滥用: launch 参数全部字面量
   854→    const b = await S.pwModule.chromium.launch({ headless: true, args: LAUNCH_ARGS })
   855→    // 浏览器重启(旧实例崩溃)时清空旧槽位, 避免持有已死 context
   856→    const stale = S.slots.splice(0, S.slots.length)
   857→    await Promise.allSettled(stale.map((s) => s.ctx.close().catch(() => {})))
   858→    S.browser = b
   859→    registerExitHooks()
   860→    // E5: 启动心跳回收定时器(幂等: 已存在则直接返回); 浏览器实例生命周期内常驻,
   861→    // shutdownObscura 时一并清理
   862→    scheduleReclaim()
   863→    return b
   864→  })()
   865→  try {
   866→    return await S.launchPromise
   867→  } finally {
   868→    S.launchPromise = null
   869→  }
   870→}

   875→    userAgent: fp.userAgent,
   876→    viewport: fp.viewport,
   877→    locale: fp.locale,

   879→    colorScheme: fp.colorScheme,
   880→    deviceScaleFactor: fp.deviceScaleFactor,
   881→    isMobile: fp.mobile,

   884→    extraHTTPHeaders: { 'Accept-Language': acceptLanguageFor(fp.locale) },

   893→  const lang = fp.locale

   907→async function createSlot(domain: string, fp: ObscuraFingerprint): Promise<PoolSlot> {
   908→  const ctx = await newStealthContext(fp)
   909→  try {
   910→    const page = await ctx.newPage()
   911→    const cdp = await applyUaCdpOverride(page, fp.userAgent)
   912→    const slot: PoolSlot = { ctx, page, domain, fp, busy: true, cdp, lastUsedAt: Date.now() }
   913→    S.slots.push(slot)
   914→    return slot
   915→  } catch (e) {
   916→    // 修复: newPage 失败时关闭已创建的 context, 防止每次失败泄漏一个空 context
   917→    try { await ctx.close().catch(() => {}) } catch { /* ignore */ }
   918→    throw e
   919→  }
   920→}

   922→/** 就地重建槽位: 关旧 context → 建新 context/page, 槽位对象引用保持不变(仍在 S.slots 中)
   923→ *  Bug4 修复(Task 4-a): 原实现先 `await slot.ctx.close()` 关旧 ctx, 再 `newStealthContext()` +
   924→ *  `ctx.newPage()` + `applyUaCdpOverride()` 全部裸跑 —— 一旦 newPage/CDP 失败, 新建的 ctx 既
   925→ *  没挂到 slot 又没 close, 每次失败泄漏一个空 BrowserContext(进程级浏览器资源, 累计会撑爆文件
   926→ *  描述符/内存)。且 slot.ctx/page/cdp 仍是已关的旧引用, 调用方拿到的 page 一访问就抛
   927→ *  "Target closed"。改为 try/catch 包裹 newPage+CDP, 失败时 await ctx.close() 回收新 ctx 再抛;
   928→ *  slot 各字段不更新(保持旧已关 ctx 引用), 调用方 withObscuraPage 的 free.page.isClosed()
   929→ *  会再次触发 recreateSlot 重建 —— 旧 ctx 已关被 recreateSlot 第一行 close().catch() 容错
   930→ *  R3-17: consecutiveFailures 计数 —— 失败 3 次后认为该槽位"不可恢复"(chromium 进程异常/
   931→ *  系统资源耗尽/page 损坏持续态), 把槽位从 S.slots 移除以缩减池容量, 避免持续重试占用
   932→ *  MAX_CONCURRENCY 名额。同时触发 checkObscuraAvailable() 重探测: 若 chromium 整体不可用,
   933→ *  probeOk 立即转 false, 后续请求直接走裸 Playwright 降级路径不再卡 obscura */
   934→async function recreateSlot(slot: PoolSlot, domain: string, fp: ObscuraFingerprint): Promise<void> {
   935→  try { await slot.ctx.close().catch(() => {}) } catch { /* ignore */ }
   936→  const ctx = await newStealthContext(fp)
   937→  try {
   938→    const page = await ctx.newPage()
   939→    const cdp = await applyUaCdpOverride(page, fp.userAgent)
   940→    slot.ctx = ctx
   941→    slot.page = page
   942→    slot.domain = domain
   943→    slot.fp = fp
   944→    slot.cdp = cdp
   945→    slot.busy = true
   946→    slot.lastUsedAt = Date.now()
   947→    // R3-17: 重建成功 → 清零连续失败计数
   948→    slot.consecutiveFailures = 0
   949→  } catch (e) {
   950→    // 新 ctx 已建但 newPage/CDP 失败: 关掉新 ctx 防泄漏, 旧 ctx 已关保持不变(下次获取时重建)
   951→    await ctx.close().catch(() => {})
   952→    // R3-17: 累计失败次数, 达阈值把槽位移出 S.slots 缩减池容量, 避免持续重试占用名额
   953→    slot.consecutiveFailures = (slot.consecutiveFailures || 0) + 1
   954→    if (slot.consecutiveFailures >= 3) {
   955→      const idx = S.slots.indexOf(slot)
   956→      if (idx >= 0) S.slots.splice(idx, 1)
   957→      console.warn(`[obscura] recreateSlot 连续失败 ${slot.consecutiveFailures} 次, 移除该槽位缩减池容量(slots=${S.slots.length})`)
   958→      // 触发重探测: 若 chromium 整体不可用, probeOk 转 false, 后续请求走降级路径
   959→      S.probeOk = null
   960→      S.probeAt = 0
   961→    }
   962→    throw e
   963→  }
   964→}

   966→function resetIdleTimer(): void {
   967→  if (S.idleTimer) clearTimeout(S.idleTimer)

   978→/**
   979→ * E5: 启动槽位级心跳回收定时器 —— 每 60s 扫描所有非 busy 槽位, 释放 10 分钟未用的 ctx
   980→ *  (槽位本身保留在 S.slots 中, 下次获取时 free.page.isClosed()=true 触发 recreateSlot 重建)。
   981→ *  补充 IDLE_CLOSE_MS(5min 全空闲关浏览器): 长任务持槽时其他空闲槽位仍可被回收,
   982→ *  避免每槽位都常驻 BrowserContext 撑资源(每个 ctx 持独立进程内堆+CDP 会话+至少 1 page)。
   983→ *  幂等: 已存在定时器则直接返回(防 ensureBrowser 多次调用累积定时器) */
   984→function scheduleReclaim(): void {
   985→  if (S.reclaimTimer) return
   986→  S.reclaimTimer = setInterval(() => {
   987→    const now = Date.now()
   988→    for (const slot of S.slots) {
   989→      // busy 槽位跳过(正在使用中, 不能关); lastUsedAt 缺失(老槽位兼容)用当前时间兜底不回收
   990→      if (slot.busy) continue
   991→      if (!slot.lastUsedAt) continue
   992→      if (now - slot.lastUsedAt < SLOT_IDLE_RECLAIM_MS) continue
   993→      // 已关的 ctx 跳过(recreateSlot 失败/上次回收后未重建)
   994→      try { if (slot.page && !slot.page.isClosed()) {
   995→        // 主动 close ctx 释放浏览器侧资源; ctx.close 自动级联关闭其所有 page,
   996→        // 槽位 page 引用随之失效, 下次 withObscuraPage 取槽时 free.page.isClosed()=true 触发 recreateSlot
   997→        void slot.ctx.close().catch(() => {})
   998→      } } catch { /* 静默 */ }
   999→    }
  1000→  }, SLOT_RECLAIM_INTERVAL_MS)
  1001→  // unref: 不阻止 CLI/一次性脚本自然退出(浏览器常驻仅在长跑服务进程中有意义)
  1002→  if (typeof S.reclaimTimer.unref === 'function') S.reclaimTimer.unref()
  1003→}

  1005→/** 进程退出/信号时尽力回收浏览器 */
  1006→function registerExitHooks(): void {
  1007→  if (S.hooksRegistered || typeof process === 'undefined') return
  1008→  S.hooksRegistered = true
  1009→  try {
  1010→    process.once('exit', () => { void shutdownObscura() })
  1011→    process.once('SIGINT', () => { void shutdownObscura().finally(() => process.exit(0)) })
  1012→    process.once('SIGTERM', () => { void shutdownObscura().finally(() => process.exit(0)) })
  1013→  } catch { /* 某些运行时只读, 忽略 */ }
  1014→}
  1015→
  1016→/** 唤醒一个排队中的等待者(信号量释放/容量空出/失败重试路径都要调, 防等待者饿死) */
  1017→function wakeNext(): void {
  1018→  const next = S.waiters.shift()
  1019→  if (next) next()
  1020→}

  1023→ * 从页面池获取一个 stealth 页面执行 fn, 用完释放不销毁。
  1024→ * - 同域名复用槽位: 保留 cookie(挑战凭证) 与指纹
  1025→ * - 跨域名复用: 销毁重建 context + 新指纹, 做好 cookie 隔离(见文件头隔离说明)
  1026→ * - 池满时排队等待(信号量语义, 并发默认 2)
  1027→ */
  1028→// ---------- 点击助手(主 frame + 跨域 iframe 全遍历, gg) ----------
  1029→/**
  1030→ * gg: clickSelector 跨域 iframe 支持(dd-d 上报引擎缺口) ——
  1031→ * page.click 只作用于主 frame, Turnstile/hCaptcha 类"复选框在 challenges.cloudflare.com
  1032→ * 跨域 iframe 内"的挑战无法交互。Playwright 的 frame 抽象原生支持跨源 frame 定位:
  1033→ * 遍历 page.frames() 逐 frame 尝试点击(首 frame=主 frame, 与原 page.click 行为一致),
  1034→ * 任一命中即返回 true; 全部未命中(含主 frame 无该元素 —— 列表页/正文页常态)返回 false,
  1035→ * 调用方保持既有"找不到静默跳过"语义不变。
  1036→ * 总闸 9s + 最多 8 frame: 防多 iframe 页面最坏 3s×N 拖慢采集; 单 frame 超时默认 3000ms
  1037→ * (与原 page.click 超时一致)。如实记录: 真实 Turnstile 无法在本沙箱验证(域内 iframe
  1038→ * 从未物化, dd-d 留档), 本增强为能力面补齐(verify-gg-a-frameclick 同构 mock 实证)。
  1039→ */
  1040→export async function clickSelectorAnywhere(page: Page, selector: string, perFrameTimeoutMs = 3000): Promise<boolean> {
  1041→  const deadline = Date.now() + 9000
  1042→  const frames = page.frames()
  1043→  for (let i = 0; i < frames.length && i < 8; i++) {
  1044→    const remain = deadline - Date.now()
  1045→    if (remain <= 200) break
  1046→    try {
  1047→      // count() 立即返回不等待: 该 frame 明确无此元素则零开销跳过
  1048→      // (不烧 perFrame 超时 —— 主 frame 无按钮时不再白等 3s)
  1049→      const cnt = await frames[i].locator(selector).count()
  1050→      if (cnt === 0) continue
  1051→      await frames[i].click(selector, { timeout: Math.min(perFrameTimeoutMs, remain) })
  1052→      return true
  1053→    } catch { /* 有元素但点击失败(被遮挡/不可点/竞态消失): 换下一个 */ }
  1054→  }
  1055→  return false
  1056→}

  1058→export async function withObscuraPage<T>(
  1059→  url: string,
  1060→  fn: (page: Page, ctx: BrowserContext) => Promise<T>,
  1061→  opts: { userAgent?: string } = {}
  1062→): Promise<T> {
  1063→  const domain = originOf(url)
  1064→  if (!domain) throw new Error(`Obscura: 无效 URL: ${url}`)
  1065→  let slot: PoolSlot | null = null
  1066→  try {
  1067→    // 信号量获取
  1068→    for (;;) {
  1069→      // 修复: 优先复用同域空闲槽位 —— 原先 find(!busy) 拿到哪个算哪个, 两个空闲槽
  1070→      // 分属不同域时跨域请求会撞掉另一个域的 context(cookie/指纹被无谓销毁重建)
  1071→      const free = S.slots.find((s) => !s.busy && s.domain === domain && !s.page.isClosed())
  1072→        ?? S.slots.find((s) => !s.busy)
  1073→      if (free) {
  1074→        free.busy = true // 先占位再异步重建, 防止并发抢占同一空槽
  1075→        // R4-12: shutdownObscura 已开始时(shuttingDown=true), 即使拿到 free 槽位也立即放弃:
  1076→        //  recreateSlot 内 newStealthContext 会因 S.browser=null 重新拉浏览器, 与 shutdownObscura
  1077→        //  正在 await b.close() 冲突, 拉起的新 ctx 会写回已被 splice 的 slot 对象(orphan)
  1078→        if (S.shuttingDown) {
  1079→          free.busy = false
  1080→          wakeNext()
  1081→          throw new Error('Obscura: 浏览器正在关闭, 请稍后重试')
  1082→        }
  1083→        if (free.domain !== domain || free.page.isClosed()) {
  1084→          try {
  1085→            await recreateSlot(free, domain, randomFingerprint({ userAgent: opts.userAgent }))
  1086→          } catch (e) {
  1087→            free.busy = false // 重建失败归还槽位(旧 ctx 已关, 下次获取时会再次重建)
  1088→            // 修复: 失败路径也必须唤醒一个等待者, 否则排队的请求会永久饥饿挂起
  1089→            wakeNext()
  1090→            throw e
  1091→          }
  1092→        }
  1093→        slot = free
  1094→        break
  1095→      }
  1096→      if (S.slots.length + S.pendingCreates < MAX_CONCURRENCY) {
  1097→        S.pendingCreates++
  1098→        try {
  1099→          slot = await createSlot(domain, randomFingerprint({ userAgent: opts.userAgent }))
  1100→        } finally {
  1101→          S.pendingCreates--
  1102→          // 建槽失败时容量已释放, 唤醒一个等待者去重试(否则等待队列可能永久挂起)
  1103→          wakeNext()
  1104→        }
  1105→        // R4-12: createSlot 完成后再次检查 shuttingDown —— shutdownObscura 在 createSlot
  1106→        //  期间(splice 全部槽位)刚发生时, 新建的 slot 已不在 S.slots 中(被 splice 清掉),
  1107→        //  ctx/page 即将 orphan。手动关闭 ctx 并抛错, 不让 fn 在已关 ctx 上跑
  1108→        if (S.shuttingDown) {
  1109→          try { await slot.ctx.close().catch(() => {}) } catch { /* ignore */ }
  1110→          slot = null
  1111→          throw new Error('Obscura: 浏览器正在关闭, 请稍后重试')
  1112→        }
  1113→        break
  1114→      }
  1115→      // R3-16: 排队等待需有 30s 超时 —— 原实现 push(resolve) 后无超时, 槽位持有者若
  1116→      // 因 page.goto 卡死(timeout 未生效/无限等待 selector)永远不释放 busy, 排队者
  1117→      // 永远卡在 await new Promise 处, 调用方 gateFetch 也跟着卡死 → 整个 obscura 路径死锁。
  1118→      // 30s 超时后 reject 触发到 gateFetch 的 catch → 落 HTTP 引擎, 链路自愈。醒来后
  1119→      // 须从 waiters 数组中把自己摘掉(否则槽位释放时 wakeNext 调用空 resolve 不报错但浪费)
  1120→      //
  1121→      // R5-18: TDZ 兜底 —— 原 setTimeout 回调引用 `resolver`, 而 `const resolver` 在
  1122→      //  setTimeout 之后声明。30s 延时下 resolver 早已赋值, 但若未来为测试改成 0ms 延时
  1123→      //  或事件循环高压下同步 fire, 会触发 TDZ ReferenceError。先 let 声明占位, 再赋值。
  1124→      let resolver: (() => void) | null = null
  1125→      await new Promise<void>((resolve, reject) => {
  1126→        const t = setTimeout(() => {
  1127→          // 从 waiters 中移除自己(若仍存在), 防 wakeNext 调到已 reject 的 resolver
  1128→          if (resolver) {
  1129→            const idx = S.waiters.indexOf(resolver)
  1130→            if (idx >= 0) S.waiters.splice(idx, 1)
  1131→          }
  1132→          reject(new Error('obscura slot timeout(30s): 池满且所有槽位长期被占用'))
  1133→        }, 30_000)
  1134→        // R5-18: 用局部 const 包装, 让 waiters.push 拿到 () => void 而非可能为 null 的引用
  1135→        const r: () => void = () => {
  1136→          clearTimeout(t)
  1137→          resolve()
  1138→        }
  1139→        resolver = r
  1140→        S.waiters.push(r)
  1141→      })
  1142→    }
  1143→    resetIdleTimer()
  1144→    return await fn(slot.page, slot.ctx)
  1145→  } finally {
  1146→    if (slot) {
  1147→      slot.busy = false
  1148→      // E5: 更新 lastUsedAt 供心跳回收定时器判定(10min 未用即 close ctx 释放资源)
  1149→      slot.lastUsedAt = Date.now()
  1150→      wakeNext()
  1151→      resetIdleTimer()
  1152→    }
  1153→  }
  1154→}

  1156→// ---------- 可用性探测 ----------
  1157→/**
  1158→ * 探测 Obscura 是否可用(成功后浏览器实例保留为单例直接复用, 不浪费 launch)。
  1159→ * 失败结果缓存 PROBE_RETRY_MS, 之后允许重新探测(例如 chromium 后装场景)。
  1160→ */
  1161→export async function checkObscuraAvailable(): Promise<boolean> {
  1162→  if (S.probeOk === true && S.browser && S.browser.isConnected()) return true
  1163→  if (S.probeOk === false && Date.now() - S.probeAt < PROBE_RETRY_MS) return false
  1164→  try {
  1165→    const b = await ensureBrowser()
  1166→    S.probeOk = b.isConnected()
  1167→  } catch (e: unknown) {
  1168→    // any 规避: 统一取 message 即可
  1169→    const msg = (e as { message?: string })?.message
  1170→    console.warn('[obscura] chromium 不可用:', (msg || String(e)).slice(0, 120))
  1171→    S.probeOk = false
  1172→  }
  1173→  S.probeAt = Date.now()
  1174→  return S.probeOk
  1175→}

  1177→// ---------- 挑战自动等待渲染 ----------
  1178→/**
  1179→ * E2: 检测 CF 挑战 UI 元素是否仍可见 —— 配合 looksLikeChallenge(HTML 内容关键词判定)
  1180→ * 双重校验, 避免仅靠 HTML 误判(JS Detections 脚本内嵌但 UI 已消失的过渡态会被关键词
  1181→ * 持续命中, 白白耗满 challengeWaitMs)。判定为"挑战仍在进行"的充分条件: ① URL 仍在
  1182→ * /cdn-cgi/challenge 路径; ② 任意挑战 UI 元素(#challenge-running / #challenge-form /
  1183→ * .cf-turnstile / challenges.cloudflare.com iframe)仍在 DOM 中存在 */
  1184→async function isChallengeUIVisible(page: Page): Promise<boolean> {
  1185→  // URL 仍在 /cdn-cgi/challenge 路径 → 挑战进行中(CF 验证页固定路径)
  1186→  try {
  1187→    if (/\/cdn-cgi\/challenge/.test(page.url())) return true
  1188→  } catch { /* page 已销毁: 视为挑战进行中, 让上层轮询继续等 */ return true }
  1189→  // 任意挑战 UI 元素仍存在 → 挑战进行中
  1190→  const selectors = [
  1191→    '#challenge-running',
  1192→    '#challenge-form',
  1193→    '.cf-turnstile',
  1194→    'iframe[src*="challenges.cloudflare.com"]',
  1195→  ]
  1196→  for (const sel of selectors) {
  1197→    try {
  1198→      const cnt = await page.locator(sel).count()
  1199→      if (cnt > 0) return true
  1200→    } catch { /* locator 失败保守视为挑战中 */ return true }
  1201→  }
  1202→  return false
  1203→}

  1205→/**
  1206→ * E2: 尝试点击 Cloudflare Turnstile 复选框(interactive 模式需要用户点击才能继续验证)。
  1207→ *  managed / non-interactive 模式 checkbox 不存在, count()=0 静默跳过; 跨域 iframe 内的
  1208→ *  checkbox 由 Playwright 的 frame 抽象跨源访问(与 clickSelectorAnywhere 同语义, gg)。
  1209→ *  主 frame 用 .cf-turnstile 限定选择器(防误点站内常规 checkbox); CF 跨域 iframe 内通常
  1210→ *  仅有一个 input[type=checkbox](即 Turnstile widget), 用通用选择器无歧义命中 */
  1211→async function tryClickTurnstile(page: Page): Promise<void> {
  1212→  // R4-13: 整体 8s 截止时间 —— 旧行为只对单 frame click 设 1500ms 超时, 但 8 个 frame
  1213→  //  最坏情况 12s, 多次调用累计接近 challengeWaitMs 40s, 浪费在 hopeless 页面上
  1214→  const deadline = Date.now() + 8000
  1215→  const frames = page.frames()
  1216→  for (let i = 0; i < frames.length && i < 8; i++) {
  1217→    if (Date.now() >= deadline) return // 整体预算耗尽, 剩余 frame 跳过(留时间给上层循环复查)
  1218→    let frameUrl = ''
  1219→    try { frameUrl = frames[i].url() || '' } catch {}
  1220→    const isCfFrame = /challenges\.cloudflare\.com|cdn-cgi\/challenge-platform/.test(frameUrl)
  1221→    // 主 frame 用 .cf-turnstile 限定(避免误点站内常规 checkbox);
  1222→    // CF 跨域 iframe 内用 input[type=checkbox](iframe 内通常仅 Turnstile 复选框)
  1223→    const sel = isCfFrame ? 'input[type=checkbox]' : '.cf-turnstile input[type=checkbox]'
  1224→    // 单 frame click 超时不得超出整体剩余预算(避免单次 click 用尽 1500ms 后整体超 8s)
  1225→    const remaining = Math.max(200, deadline - Date.now())
  1226→    try {
  1227→      const cnt = await frames[i].locator(sel).count()
  1228→      if (cnt === 0) continue
  1229→      await frames[i].click(sel, { timeout: Math.min(1500, remaining) })
  1230→      return
  1231→    } catch { /* 静默: 元素消失/被遮挡/不可点 — 换下一个 frame */ }
  1232→  }
  1233→}

  1235→/**
  1236→ * --stealth 模式渲染:
  1237→ * 1. goto(domcontentloaded) → 读 content
  1238→ * 2. 命中挑战特征 → 每 1~3s 轮询 content+url+UI, 最多 challengeWaitMs(默认 40s), 通过即止
  1239→ *    E2 增强: ① 1-3s 人类延迟 + Turnstile 复选框点击(interactive 模式需用户交互)
  1240→ *           ② 结构化消失判定(#challenge-running/.cf-turnstile 等元素消失 + URL 不含 /cdn-cgi/challenge)
  1241→ *           ③ 超时不再抛错 —— 返回当前页面状态, 由 fetcher.looksBlocked 检测并降级;
  1242→ *              obscura 期间拿到的 cf_clearance 等 cookie 仍会写入 CookieJar 供 HTTP 引擎直连复用
  1243→ * 3. waitSelector(容忍超时)/waitMs 支持
  1244→ * 4. 全量回传 context.cookies() 为 Set-Cookie 风格字符串数组
  1245→ */
  1246→export async function renderStealth(url: string, opts: ObscuraFetchOptions = {}): Promise<ObscuraFetchResult> {
  1247→  const timeout = opts.timeout ?? 20000
  1248→  // Turnstile 管理型挑战(无交互自动验证)常需 10~25s, 慢尾可达 35s+, 默认 40s 上限
  1249→  const challengeWaitMs = opts.challengeWaitMs ?? 40000
  1250→  return withObscuraPage(url, async (page, ctx) => {
  1251→    try {
  1252→      await page.goto(url, {
  1253→        waitUntil: 'domcontentloaded',
  1254→        timeout,
  1255→        ...(opts.referer ? { referer: opts.referer } : {}),
  1256→      })
  1257→    } catch (e: any) {
  1258→      // 修复: CF 管理型挑战会扣住导航直至验证通过, goto 常在 domcontentloaded 前超时,
  1259→      // 但页面其实活着且后续会自动 reload —— 只有超时类错误才落入下方挑战等待循环抢救,
  1260→      // 其余(DNS/连接拒绝)照常抛出; about:blank 空页也不抢救防把空壳当成功内容
  1261→      const msg = String(e?.message || e)
  1262→      if (!/timeout|timed?\s?out/i.test(msg) || page.url() === 'about:blank') throw e
  1263→    }
  1264→    let html = await page.content()
  1265→    let challengeWaited = false
  1266→
  1267→    // —— 挑战等待循环: CF 等盾会自动验证并刷新页面; E2 增强:
  1268→    //  ① 1-3s 人类随机延迟(模拟阅读挑战提示后操作, 防被识别为机器行为)
  1269→    //  ② 尝试点击 Turnstile 复选框(interactive 模式需要用户交互才能继续验证)
  1270→    //  ③ 结构化消失判定(#challenge-running/.cf-turnstile 等元素消失 + URL 不含 /cdn-cgi/challenge)
  1271→    //  ④ 超时不再抛错 —— 返回当前页面状态, 由 fetcher.looksBlocked 检测并降级
  1272→    const start = Date.now()
  1273→    while (looksLikeChallenge(html) && Date.now() - start < challengeWaitMs) {
  1274→      challengeWaited = true
  1275→      // E2 ①: 1-3s 随机延迟(原实现固定 1s 轮询节奏过于规律, 易被 CF 行为分析识别)
  1276→      await page.waitForTimeout(1000 + Math.floor(Math.random() * 2000)).catch(() => {})
  1277→      // E2 ②: 尝试点击 Turnstile 复选框(interactive 模式无 checkbox 时静默跳过)
  1278→      await tryClickTurnstile(page).catch(() => {})
  1279→      // E2 ③: 检查挑战 UI 是否结构化消失(UI 消失但 HTML 关键词仍命中的过渡态可提前出循环)
  1280→      let uiGone = false
  1281→      try { uiGone = !(await isChallengeUIVisible(page)) } catch { uiGone = false }
  1282→      if (uiGone) {
  1283→        // UI 消失后等 DOM 稳定再取新内容(盾通过后页面常自动 reload)
  1284→        await page.waitForLoadState('domcontentloaded', { timeout: 5000 }).catch(() => {})
  1285→      }
  1286→      try {
  1287→        html = await page.content()
  1288→      } catch {
  1289→        // 修复: 盾通过时页面正在 reload, content() 会抛 "Execution context destroyed";
  1290→        // 置空让 isJsRedirectShell('')=true 继续轮询, 下一轮即可拿到真实页面
  1291→        html = ''
  1292→      }
  1293→      // UI 已消失且内容不再像挑战 → 提前退出循环(无需耗满 challengeWaitMs)
  1294→      if (uiGone && !looksLikeChallenge(html)) break
  1295→    }
  1296→    if (challengeWaited) {
  1297→      // 盾通过后页面通常自动 reload, 等 DOM 稳定再取内容
  1298→      await page.waitForLoadState('domcontentloaded', { timeout: 8000 }).catch(() => {})
  1299→      await page.waitForTimeout(500)
  1300→      try { html = await page.content() } catch { /* 保持上一轮内容 */ }
  1301→    }
  1302→
  1303→    // E2 ④: 挑战超时不再抛错 —— 返回当前页面状态(可能仍是挑战页), 由 fetcher 的
  1304→    // looksBlocked 检测并降级; obscura 期间拿到的 cf_clearance 等挑战凭证 cookie 仍会
  1305→    // 写入 CookieJar(fetcher.renderWithBrowser: cookieJar.store(originHost(url), res.cookies)),
  1306→    // 后续 HTTP 引擎直连可凭其过盾 —— 原 throw 会丢失这部分关键凭证, 让降级裸 Playwright
  1307→    // 也过不了盾。waitSelector/settle/click 等下游处理仍正常进行, 或许能进一步加载出真内容
  1308→
  1309→    if (opts.waitSelector) {
  1310→      await page.waitForSelector(opts.waitSelector, { timeout: opts.waitMs || 8000 }).catch(() => { /* 容忍超时 */ })
  1311→    }
  1312→    if (opts.waitMs) await page.waitForTimeout(opts.waitMs)
  1313→
  1314→    // —— 点击展开懒加载内容(ixdzs/101kks 系"点击展开全部目录"交互)——
  1315→    // AJAX 预取了数据但只有点击才注入 DOM; gg: 主 frame+跨域 iframe 全遍历
  1316→    // (Turnstile/hCaptcha 类挑战复选框在跨域 iframe 内, dd-d 缺口补齐);
  1317→    // 找不到元素(列表页/正文页)时静默跳过(全 frame 未命中=false → 跳过, 语义不变)
  1318→    if (opts.clickSelector) {
  1319→      const clicked = await clickSelectorAnywhere(page, opts.clickSelector)
  1320→      if (clicked) await page.waitForTimeout(1200)
  1321→    }
  1322→
  1323→    // —— 渲染稳定化采样: AJAX 页面在 waitMs 后仍可能持续注入内容(章节列表延迟加载、
  1324→    // 段落客户端组装等), 每 700ms 采样 HTML 尺寸, 连续 2 次增长 <0.5% 视为稳定;
  1325→    // 最多再等 settleMs(默认 6000), 稳定后提前退出避免无谓延迟
  1326→    const settleMs = opts.settleMs ?? 6000
  1327→    if (settleMs > 0) {
  1328→      const settleStart = Date.now()
  1329→      let lastLen = html.length
  1330→      try { lastLen = (await page.content()).length } catch { /* 导航中: 用上一轮长度 */ }
  1331→      let stable = 0
  1332→      while (Date.now() - settleStart < settleMs && stable < 2) {
  1333→        await page.waitForTimeout(700)
  1334→        let len = lastLen
  1335→        try { len = (await page.content()).length } catch { break /* 页面导航中 */ }
  1336→        if (Math.abs(len - lastLen) <= Math.max(64, lastLen * 0.005)) stable++
  1337→        else stable = 0
  1338→        lastLen = len
  1339→      }
  1340→    }
  1341→
  1342→    let finalHtml = html
  1343→    // 点击"展开"按钮可能实为链接触发整页导航: content() 在导航提交期间会抛
  1344→    // "Execution context destroyed", 单次失败退回上一轮采样会把点击前的空目录当最终结果;
  1345→    // 短退避重试拿导航后的真实内容, 重试耗尽仍失败才退回上一轮
  1346→    for (let retry = 0; retry < 3; retry++) {
  1347→      try { finalHtml = await page.content(); break } catch {
  1348→        if (retry === 2) break
  1349→        try { await page.waitForTimeout(700) } catch { break /* 页面已销毁 */ }
  1350→      }
  1351→    }
  1352→
  1353→    // E2 ④(续): 渲染最终结果仍为挑战页(如二次盾)也不再抛错 —— 与挑战超时同策略,
  1354→    // 由 fetcher 的 looksBlocked 检测并降级处理(blocked=true 走 cookie 重试/裸 Playwright);
  1355→    // obscura 期间拿到的 cf_clearance 等凭证 cookie 仍会写入 CookieJar 供后续 HTTP 引擎直连复用
  1356→    // if (looksLikeChallenge(finalHtml)) { /* 不抛: 交 fetcher.looksBlocked 处理 */ }
  1357→
  1358→    // —— Cookie 回传: 转为 Set-Cookie 风格字符串, 供 fetcher 写入 CookieJar ——
  1359→    // Bug18 修复(Task 4-a): 原实现按 `new URL(url).hostname` 过滤 ctx.cookies(), 用的是【请求
  1360→    // URL】而非【page.url() 最终URL】—— 挑战通过后页面常发生同 host 跳转(如 /cdn-cgi/challenge
  1361→    // → /), 但跨子域跳转(www.example.com → example.com)或重定向链会令 host 不匹配, 关键的
  1362→    // cf_clearance / sessionId 等 cookie 会被 domainMatch 误删, 回传给 fetcher 时已无挑战凭证,
  1363→    // 后续 HTTP 直连仍过盾失败。改用更稳健策略: 全量回传 ctx.cookies(), 由 fetcher 侧
  1364→    // cookieJar.store(originHost(url), ...) 按请求 URL host 分桶存储 —— 同源 cookie 自然落入目标
  1365→    // 域罐, 第三方 cookie 即便被存入也只是无害冗余(目标站不识别即忽略, 不会被回送原第三方域);
  1366→    // 关键凭证 cf_clearance 零丢失, HTTP 引擎后续直连可凭其过盾。Obscura 槽位本就按 origin
  1367→    // 隔离(同域复用/跨域重建), 第三方 cookie 在槽位生命周期内本就少且短命
  1368→    const cookies = await ctx.cookies()
  1369→    const cookieStrings = cookies.map((c) => {
  1370→      const parts = [`${c.name}=${c.value}`, `path=${c.path || '/'}`, `domain=${c.domain}`]
  1371→      if (c.secure) parts.push('Secure')
  1372→      if (c.httpOnly) parts.push('HttpOnly')
  1373→      return parts.join('; ')
  1374→    })
  1375→
  1376→    return {
  1377→      html: finalHtml,
  1378→      cookies: cookieStrings,
  1379→      finalUrl: page.url(),
  1380→      challengeWaited,
  1381→    }
  1382→  }, { userAgent: opts.userAgent })
  1383→}

  1386→/** Obscura 抓取: 不可用时抛错(由调用方决定降级策略) */
  1387→export async function obscuraFetch(url: string, opts: ObscuraFetchOptions = {}): Promise<ObscuraFetchResult> {
  1388→  const ok = await checkObscuraAvailable()
  1389→  if (!ok) throw new Error('Obscura 不可用: chromium 未安装或启动失败')
  1390→  return renderStealth(url, opts)
  1391→}

  1393→/** 主动关闭浏览器与页面池(空闲回收/进程退出/测试收尾时调用) */
  1394→export async function shutdownObscura(): Promise<void> {
  1395→  // R4-12: 设置 shuttingDown 标志, withObscuraPage 获取/重建槽位路径会检测本标志并立即抛错,
  1396→  //  防止 shutdownObscura splice 全部槽位后又有新槽位被创建(orphan ctx/page 泄漏)
  1397→  S.shuttingDown = true
  1398→  if (S.idleTimer) { clearTimeout(S.idleTimer); S.idleTimer = null }
  1399→  // E5: 取消心跳回收定时器(浏览器实例已关, 不再需要扫描)
  1400→  if (S.reclaimTimer) { clearInterval(S.reclaimTimer); S.reclaimTimer = null }
  1401→  const slots = S.slots.splice(0, S.slots.length)
  1402→  await Promise.allSettled(slots.map((s) => s.ctx.close().catch(() => {})))
  1403→  const b = S.browser
  1404→  S.browser = null
  1405→  S.probeOk = null
  1406→  S.probeAt = 0
  1407→  // 唤醒所有等待者: 让它们重新评估(浏览器已关, 按需重建或快速失败), 防止永久挂起
  1408→  const waiters = S.waiters.splice(0, S.waiters.length)
  1409→  for (const w of waiters) w()
  1410→  if (b) {
  1411→    try { await b.close() } catch { /* 已死则忽略 */ }
  1412→  }
  1413→  // 清除 shuttingDown 标志(收尾完成, 后续 ensureBrowser 重新拉起时不再被拦截)
  1414→  S.shuttingDown = false
  1415→}
