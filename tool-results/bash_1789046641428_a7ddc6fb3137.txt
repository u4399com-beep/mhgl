     7→/**
     8→ * 字段提取规则
     9→ *
    10→ * JSON 站点(json/const 类型)专用语法约定(与 parser.ts 实现严格对齐):
    11→ *
    12→ * type='json': expression 为 JSON 点路径, 从响应体(JSON.parse 后)取值 ——
    13→ *   - 点分逐层: `a.b.c` → data.a.b.c; 数字段=数组下标(0基): `items.0.title`
    14→ *   - 空路径或 `.` → 当前根值本身(数组项为纯字符串时用, 如纯章节名数组)
    15→ *   - `[]` 为装饰可剔: `hotlist[].id` ≡ `hotlist.id`
    16→ *   - 首段为空按根处理: 根数组取 `.0.title` ≡ `[0].title`
    17→ *   - 取值结果: 数组→各元素转字符串按\n连接; 标量→String; null/undefined/对象→''
    18→ *   - cc-c 扩展(加法语义, 既有路径零回归):
    19→ *     `[n]` 段内方括号下标: `tabs[0].title` ≡ `tabs.0.title`
    20→ *     `[k=v]` 过滤算子: `search_tabs[tab_type=3]` 保留元素属性值(可 & 连写多条件,
    21→ *           值按 String 宽松比较, 数字/字符串通吃); 非数组值上为无操作
    22→ *     `*` 段: 数组递归展平(数组的数组→元素平面): `chapterListWithVolume.*` → 章节数组
    23→ *
    24→ * type='const': expression 为常量模板字符串, `{name}` 占位符替换后作为字段值 ——
    25→ *   - `{字段名}`  → 同一作用域已提取的同名字段值(列表项/书籍页字段, 如 `{id}`)
    26→ *   - `{index}`   → 1基序号(列表项/目录数组项遍历时; 章节API chapterid=下标+1 场景直接可用)
    27→ *   - `{q.参数名}` → 当前页面 URL 的查询参数(如书页URL /api/book?id=2530 → `{q.id}`)
    28→ *   - 未命中的占位符替换为空串; 后处理(replaceFrom/stripTags等)照常生效
    29→ *   - 典型: bookUrl=`https://x/api/book?id={id}`; 章节URL=`https://x/api/chapter?id={q.id}&chapterid={index}`
    30→ *
    31→ * itemSelector.type='json' 时 expression 指向"数组路径": 支持逗号分隔多路径并集
    32→ * (`hotlist,sort1,uplist` → 各数组顺序拼接, 适配首页多榜单API), 对数组每项跑 fields 提取。
    33→ * cc-c 扩展: 数组路径另支持 map-collect(非数字段作用在数组上=跨元素取该属性并展平一层,
    34→ * 三层嵌套一次下钻: `search_tabs[tab_type=3].data.book_data`)与 `*` 递归展平段
    35→ * (数组的数组→章节平面: `chapterListWithVolume.*`), 语法与 jsonGet 同一契约。
    36→ */
    37→export interface FieldRule {
    38→  /** 提取方式 (json/const 语法见接口头注释) */
    39→  type: 'css' | 'xpath' | 'regex' | 'json' | 'const'
    40→  /** css: 选择器 | xpath: 表达式 | regex: 正则 | json: JSON点路径 | const: 常量模板 */
    41→  expression: string
    42→  /** css/xpath: 取值方式 text/html/href/src/属性名; regex: 捕获组序号; json/const: 无意义 */
    43→  attr?: string
    44→  /** regex 使用标志 (默认 gis) */
    45→  flags?: string
    46→  /** 后处理: 是否剔除HTML标签 */
    47→  stripTags?: boolean
    48→  /** 后处理: 替换规则 源→目标 (支持正则) */
    49→  replaceFrom?: string
    50→  replaceTo?: string
    51→  /** 后处理: 截取第N项(逗号分隔结果) */
    52→  index?: number
    53→}

    81→    joinWith?: string

    86→export interface FetchConfig {
    87→  /** auto: HTTP失败自动切浏览器渲染; http: 纯HTTP; browser: 强制浏览器 */
    88→  engine: 'auto' | 'http' | 'browser'
    89→  /** ua: rotate 每次随机UA | fixed 固定随机一次 | custom 自定义UA |
    90→   *  mobile 移动UA子集(池内 iPhone/iPad/Android, 同域钉扎) |
    91→   *  desktop 桌面UA子集(池内 Windows/Mac/Linux, 同域钉扎)。
    92→   *  mobile/desktop 两档会按所选 UA 自动配套完整浏览器指纹头组
    93→   *  (sec-ch-ua/sec-ch-ua-mobile/sec-ch-ua-platform/Sec-Fetch-*, ff-b),
    94→   *  rotate/fixed/custom 亦按实际命中的 UA 自动推导(移动 UA 配移动平台指纹) */
    95→  uaMode: 'rotate' | 'fixed' | 'custom' | 'mobile' | 'desktop'
    96→  /** uaMode=fixed 时使用 pool 内随机, =custom 时使用 customUa */
    97→  customUa?: string
    98→  /** 附加请求头 */
    99→  headers?: Record<string, string>
   100→  /** Cookie 字符串 (k=v; k2=v2) */
   101→  cookies?: string
   102→  /** 是否自动跟随 Set-Cookie */
   103→  autoCookie?: boolean
   104→  /** 是否携带 Referer (自动为站点首页) */
   105→  referer?: boolean
   106→  /** Referer 链伪造(ff-b): 开启后各请求优先携带"来源页"Referer 而非站点首页 origin——
   107→   *  目录页带书籍页 Referer、章节页带目录页 Referer、翻页带当前页 Referer
   108→   *  (很多站校验 Referer 同域/同链路, 固定 origin 的裸 Referer 本身是爬虫指纹)。
   109→   *  来源页由 runner/parser 运行时经 refererUrl 逐请求注入; 未注入时回退 origin(零回归) */
   110→  refererChain?: boolean
   111→  /** 当前请求的来源页 URL(refererChain 启用时作 Referer; 运行时逐请求注入, 同 pageFetch
   112→   *  属运行时项 —— sanitizeFetchConfig 白名单不透传, 规则 JSON 同名键自然丢弃) */
   113→  refererUrl?: string
   114→  /** 超时 ms */
   115→  timeout: number
   116→  /** 失败重试次数 */
   117→  retries: number
   118→  /** 浏览器渲染等待选择器(可选) */
   119→  waitSelector?: string
   120→  /** 浏览器渲染等待 ms */
   121→  waitMs?: number
   122→  /** 浏览器渲染后需要点击以展开懒加载内容的选择器(如"点击展开全部目录");
   123→   *  页面无该元素时静默跳过, 适配 AJAX 预取+点击注入型目录站 */
   124→  clickSelector?: string
   125→  /** 触发浏览器渲染的状态码 */
   126→  browserFallbackStatus?: number[]
   127→  /** 同站并发闸门(hostGate)在飞上限, 缺省 3; 1~10 钳制(sanitizeFetchConfig 同步);
   128→   *  同 host 连续失败自动降额(最低1)/连续成功回升(不超过此基准) */
   129→  hostGateLimit?: number
   130→  /** 翻页请求传输回调(可选, 运行时注入): parseToc/parseContent 内部"下一页"抓取的传输实现。
   131→   *  runner 注入过闸版 gateFetch(翻页请求与章节抓取同享同站并发闸, aa-f 已知边界闭环);
   132→   *  未注入时 parser 直连 fetchPage —— rules/test 测试路由保持直连语义不变。
   133→   *  注: 仅运行时注入项, sanitizeFetchConfig 白名单不透传(函数无法 JSON 序列化,
   134→   *  规则 JSON 中的同名键会被白名单自然丢弃, 无注入面) */
   135→  /** 翻页请求注入回调(bb-d, 运行时项): runner 侧过闸抓取。ll-c 扩展可选第二参 refererUrl
   136→   *  —— parseToc/parseContent 翻页第2页起回传上一页 URL, 启用 refererChain 的调用方可把
   137→   *  Referer 从"恒书籍页"升级为"翻页链逐页回溯"(真实浏览器翻页导航语义) */
   138→  pageFetch?: (url: string, refererUrl?: string) => Promise<{ html: string }>
   139→  /** 通用 token 预取钩子(bb-d): 请求前先从 tokenUrl 预取动态 token 再注入请求。
   140→   *  面向"可预取 token"形态(会话级/短时级); 按章变化的加密参数型(如 bqg713 AES)
   141→   *  不属此形态, 需站点专属解密或外置转换代理(tokenUrl 可用 {url} 占位符对接) */
   142→  /** 预取地址: 响应体含 token 的任意端点(通常 JSON API); 支持 {url} 占位符=
   143→   *  当前请求 URL 的 encodeURIComponent(外部转换代理形态) */
   144→  tokenUrl?: string
   145→  /** 提取表达式: 'regex:' 前缀=正则(取第一捕获组, 无捕获组取全匹配), 否则按 JSON 点路径
   146→   *  (语法同 parser.jsonGet, 如 'data.token') */
   147→  tokenPattern?: string
   148→  /** 注入方式: 'url'=替换请求 URL 中 {token}/%7Btoken%7D 占位符(无占位符时追加
   149→   *  ?token=/&token= 查询参数); 'header'=注入请求头 tokenHeaderName(默认 X-Token) */
   150→  tokenInjection?: 'url' | 'header'
   151→  /** tokenInjection='header' 时的请求头名, 缺省 X-Token */
   152→  tokenHeaderName?: string
   153→  /** 内容代理 URL(如 xjp-proxy /content?u={url}): 配置后, content 段的 fetch 不走原始 URL,
   154→   *  而是请求 contentProxyUrl(替换 {url} 为原始章节 URL 的 encodeURIComponent), 代理返回
   155→   *  JSON {ok:true, content:string} 或 {ok:false, error:string}, 引擎直接用该 content 作为 HTML
   156→   *  传给 parser, 不再 fetch 原始站点。适用于 var c / AES 加密等需服务端解密的站点
   157→   *  (xjp-proxy/deqixs-proxy 等)。代理响应体为纯文本(以 \n 分段), 引擎把每行 wrap 成 <p>
   158→   *  传给 parser(content 字段可设 const 类型直接拿全文)。代理失败/响应非法时降级直连原 URL
   159→   *  (零回归兜底)。loopback(127.0.0.1:301x 等)经 assertSafeTarget({allowLoopback:true}) 放行 */
   160→  contentProxyUrl?: string
   161→  /** 出口代理池(dd-a, 反反爬核心): 逗号分隔多条代理, 如
   162→   *  "http://u:p@host:port,http://host2:port2"; 多条时每请求随机轮换(与 UA 池同款
   163→   *  random 模式, 分布可测试验证); 失败按池逐条重试, 全部失败降级直连重试一次。
   164→   *  目标为回环地址(localhost/127.x/::1)时自动豁免直连(本地 mock/token 代理
   165→   *  tokenUrl 经代理转发会出不去)。支持矩阵(fetcher.ts 代理池段实测记录):
   166→   *  bun 运行时 fetch 仅 http/https(socks5 落 curl 链); node 运行时(next dev/prod
   167→   *  实测以 node 运行)fetch 静默忽略 proxy → 代理尝试直接走 curl 链(-x 全形态);
   168→   *  裸 Playwright per-context 全形态/Obscura 路径不支持(配置代理时自动改走裸 Playwright) */
   169→  proxyUrl?: string
   170→  /** 镜像域名自动故障切换(dd-b): 逗号分隔域名列表(可选带端口), 顺序=优先级, 如
   171→   *  "apibi.cc,apiqu.cc,apige.cc"。语义: URL 自身 host + 本列表全部视为同一镜像组,
   172→   *  请求失败且错误为网络层/超时/403/5xx 时按组序重写 host 重试(从下一镜像起, 至多组大小次);
   173→   *  404/2xx/3xx 不触发(404=资源不存在换镜像无意义, 存档裁定)。重写后 URL 走完整
   174→   *  fetch 流程(代理池/回环豁免/UA/Cookie 逻辑照常), token 预取 {url} 占位符自动拿到
   175→   *  重写后 URL → 逐章 token 天然按镜像域重签; 镜像重试在 transport 层不经 hostGate 闸门 */
   176→  mirrorDomains?: string
   177→  /** 代理轮换策略(feat-round-8: Feature B3):
   178→   *  - 'round-robin' (缺省): 按池顺序轮换, 分布均匀, 与既有 fetchHttpWithCurlFallback 的
   179→   *    Fisher-Yates 洗牌相近(实际仍按 useCount 取池中未失败条目顺序选);
   180→   *  - 'random': 每次从可用池中随机选一条(原 pickProxyFor 行为);
   181→   *  - 'least-used': 跟踪每条代理 useCount, 选使用次数最少的(平摊负载, 适合长任务);
   182→   *  失败冷却: 任一代理超时/连接错误时打 failed 标记 + 30s 冷却, 轮换时跳过冷却中的代理;
   183→   *  冷却过期后自动恢复参与轮换(无需手动重置)。运行时状态(useCount/failedUntil)
   184→   *  在 fetcher.ts 进程级 Map 持久, 不进规则 JSON(sanitize 白名单不透传) */
   185→  proxyRotationStrategy?: 'round-robin' | 'random' | 'least-used'
   186→  /** 请求抖动(feat-round-8: Feature B1): 0~N ms 的随机抖动叠加在 runner 批次间隔上,
   187→   *  让请求节奏更不规则, 防简单 rate-pattern 检测。runner 在每批 gateFetch 前 sleep
   188→   *  jitterMs(随机 0~jitterMs); 缺省 0=关闭(零回归); 引擎层不直接消费此字段,
   189→   *  由 runner 读 cfg.fetch.jitterMs 在批次循环前 sleep */
   190→  jitterMs?: number
   191→  /** 采集传输模式(hh-c, 第三方抓取工具接入): 'native'(缺省)=既有引擎链路(bun
   192→   *  fetch/curl/Obscura 全家桶, 零回归); 'scrapling-static'=经 scrapling-bridge 静态
   193→   *  传输(curl_cffi TLS 指纹伪装+浏览器头组); 'scrapling-stealthy'=经 scrapling-bridge
   194→   *  隐身浏览器(patchright 反检测+CF 挑战自动求解); 'scrapling-playwright'=经
   195→   *  scrapling-bridge 裸 Playwright chromium JS 渲染。
   196→   *  scrapling-* 语义: 整次抓取交本机桥服务代发, 目标侧响应(含 4xx/5xx)如实返回 ——
   197→   *  token 预取/autoCookie/Cookie 挑战重试/浏览器升级链等 native 专有步骤跳过(隐身能力
   198→   *  由桥内 Scrapling Fetcher 自身承担); 桥不可达或桥内异常时降级既有 native 链一次。
   199→   *  引擎侧对非法值(非白名单枚举)一律回退 native(sanitize 白名单枚举校验 + fetcher
   200→   *  scraplingModeOf 双重防线); 与 mirrorDomains 组合时逐镜像 host 各走一次桥分流 */
   201→  fetchMode?: string
   202→  /** scrapling 桥地址(fetchMode=scrapling-* 时生效), 缺省 http://127.0.0.1:3012
   203→   *  (可用环境变量 SCRAPLING_BRIDGE_URL 改全局缺省); 桥服务见 mini-services/scrapling-bridge */
   204→  scraplingBridgeUrl?: string
   205→}

   303→export function parseRuleConfig(raw: string | null | undefined): RuleConfig {
   304→  const base = defaultRuleConfig()
   305→  if (!raw) return base
   306→  try {
   307→    const cfg = JSON.parse(raw)
   308→    // 防脏数据: JSON 根不是纯对象(数组/数字/字符串)时直接回退默认配置,
   309→    // 避免展开数组把 "0"/"1" 数字索引键扩散进结果对象
   310→    if (!cfg || typeof cfg !== 'object' || Array.isArray(cfg)) return base
   311→    // 深消毒: 各段逐字段白名单重建(脏 JSON 序列化残留/字符串数字/未知键全部剔除),
   312→    // 与实采 fetcher/解析器期望的形状严格对齐 —— 浅合并会让 cfg.fetch 里的未知键
   313→    // (如运行时对象误存)与脏类型直通引擎
   314→    const sFetch = sanitizeFetchConfig(cfg.fetch)
   315→    const sClean = sanitizeCleanConfig(cfg.clean)
   316→    const sList = sanitizePageRule(cfg.list)
   317→    const sBook = sanitizePageRule(cfg.book)
   318→    const sToc = sanitizePageRule(cfg.toc)
   319→    const sContent = sanitizePageRule(cfg.content)
   320→    return {
   321→      ...base,
   322→      fetch: { ...base.fetch, ...sFetch },
   323→      clean: sClean ?? base.clean,
   324→      list: sList ?? base.list,
   325→      book: sBook ?? base.book,
   326→      toc: sToc ?? base.toc,
   327→      content: sContent ?? base.content,
   328→    }
   329→  } catch {
   330→    return base
   331→  }
   332→}

   337→// 字符串拼接("3"+1="31" 次)、fetch 参数 TypeError 等真实缺陷。此处以
   338→// "逐字段白名单+钳制"重建, 未知键全部丢弃, 供 runner 实采链路与

   342→/** 安全数字: 非数字/NaN/Infinity 返回 undefined; 字符串数字("20000")按 Number 转换 */
   343→function safeNum(v: unknown, min: number, max: number): number | undefined {
   344→  const n = typeof v === 'number' ? v : typeof v === 'string' && v.trim() !== '' ? Number(v) : NaN
   345→  if (!Number.isFinite(n)) return undefined
   346→  return Math.min(max, Math.max(min, Math.round(n)))
   347→}
   348→
   349→/** 安全布尔: 仅接受真布尔(防 "false" 字符串恒真) */
   350→function safeBool(v: unknown): boolean | undefined {
   351→  return typeof v === 'boolean' ? v : undefined
   352→}
   353→
   354→/** 安全字符串: 非字符串丢弃, 钳长度 */
   355→function safeStr(v: unknown, max: number): string | undefined {
   356→  if (typeof v !== 'string') return undefined
   357→  const s = v.slice(0, max)
   358→  return s
   359→}
   360→
   361→/**
   362→ * 单行头值安全化(ee-d): 剥除 CR/LF/NUL 等控制字符 —— 防护面:
   363→ * ① 请求头/UA/Cookie 注入拼接(HTTP 请求走私/断链) ② 入库规则被测试路由/引擎双出口消费,
   364→ * 修复必须在 sanitize 层而非仅 fetch 构造层(defense in depth, bb 轮 fetch 层修复的补全面)。
   365→ * 保留空格化语义: CR/LF 置空而非丢弃前后文(与 bb 轮 fetch 层同口径), NUL 直接剥除。
   366→ */
   367→function safeSingleLine(v: string): string {
   368→  return v.replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, '').replace(/\r\n?|\n/g, ' ')
   369→}
   370→
   371→/** HTTP 头键安全化(ee-d): 仅保留 RFC 7230 token 字符, 非法键返回 undefined(丢弃)
   372→ *  R4-22: 同时拒收 HTTP smuggling 向量头(host/content-length/transfer-encoding/
   373→ *  connection/upgrade/te/trailer/expect)——这些头由运行时/代理统一管理, 用户配置注入
   374→ *  会引发请求走私/分块解析混淆/连接复用串味。仅放行真正应用层自定义头
   375→ *  R5-15: 追加 via / x-forwarded-* / x-real-ip / forwarded / x-original-url /
   376→ *  x-rewrite-url / x-cluster-client-ip —— 这些是 Caddy(见 Caddyfile)/反代/WAF 写入的
   377→ *  代理识别头, 上游服务可能据此做 IP 鉴权; 引擎再发同名头会与代理头冲突或绕过 IP 鉴权 */
   378→const HEADER_KEY_DENYLIST = new Set([
   379→  'host', 'content-length', 'transfer-encoding', 'connection', 'upgrade',
   380→  'te', 'trailer', 'expect', 'keep-alive', 'proxy-connection', 'proxy-authorization',
   381→  'proxy-authenticate', 'front-end-https', 'x-http-method-override',
   382→  // R5-15: 代理识别/转发链头
   383→  'via', 'forwarded',
   384→  'x-forwarded-for', 'x-forwarded-host', 'x-forwarded-proto', 'x-forwarded-port',
   385→  'x-forwarded-server', 'x-real-ip', 'x-original-url', 'x-rewrite-url',
   386→  'x-cluster-client-ip',
   387→])
   388→function safeHeaderKey(v: string): string | undefined {
   389→  const s = v.replace(/[^!#$%&'*+\-.^_`|~0-9A-Za-z]/g, '')
   390→  if (s.length === 0) return undefined
   391→  // R4-22: 拒收 smuggling 向量头(大小写不敏感)
   392→  if (HEADER_KEY_DENYLIST.has(s.toLowerCase())) return undefined
   393→  return s
   394→}

   396→/** 安全字符串数组: 过滤非字符串项, 钳项数与单项长度 */
   397→function safeStrArr(v: unknown, maxItems: number, maxLen: number): string[] | undefined {
   398→  if (!Array.isArray(v)) return undefined
   399→  const out = v
   400→    .filter((x): x is string => typeof x === 'string')
   401→    .slice(0, maxItems)
   402→    .map((x) => x.slice(0, maxLen))
   403→  return out
   404→}

   406→/** FieldRule 消毒: type 白名单/表达式/属性/后处理参数逐字段重建 */
   407→export function sanitizeFieldRule(v: unknown): FieldRule | undefined {
   408→  if (!v || typeof v !== 'object' || Array.isArray(v)) return undefined
   409→  const r = v as Record<string, unknown>
   410→  const type = r.type
   411→  if (type !== 'css' && type !== 'xpath' && type !== 'regex' && type !== 'json' && type !== 'const') return undefined
   412→  const expression = safeStr(r.expression, 1000)
   413→  if (expression === undefined || !expression.trim()) return undefined
   414→  const out: FieldRule = { type, expression }
   415→  const attr = safeStr(r.attr, 100)
   416→  if (attr !== undefined) out.attr = attr
   417→  const flags = safeStr(r.flags, 10)
   418→  if (flags !== undefined) out.flags = flags
   419→  const replaceFrom = safeStr(r.replaceFrom, 1000)
   420→  if (replaceFrom !== undefined) out.replaceFrom = replaceFrom
   421→  const replaceTo = safeStr(r.replaceTo, 1000)
   422→  if (replaceTo !== undefined) out.replaceTo = replaceTo
   423→  const stripTags = safeBool(r.stripTags)
   424→  if (stripTags !== undefined) out.stripTags = stripTags
   425→  const index = safeNum(r.index, 0, 100)
   426→  if (index !== undefined) out.index = index
   427→  return out
   428→}

   430→/** PageRule 消毒: 四段页面规则(itemSelector/tocLink/fields/pagination)白名单重建 */
   431→export function sanitizePageRule(v: unknown): PageRule | undefined {
   432→  if (!v || typeof v !== 'object' || Array.isArray(v)) return undefined
   433→  const r = v as Record<string, unknown>
   434→  const enabled = safeBool(r.enabled)
   435→  const out: PageRule = { enabled: enabled ?? true, fields: {} }
   436→  const urlTemplate = safeStr(r.urlTemplate, 1000)
   437→  if (urlTemplate !== undefined) out.urlTemplate = urlTemplate
   438→  const itemSelector = sanitizeFieldRule(r.itemSelector)
   439→  if (itemSelector) out.itemSelector = itemSelector
   440→  const tocLink = sanitizeFieldRule(r.tocLink)
   441→  if (tocLink) out.tocLink = tocLink
   442→  if (r.fields && typeof r.fields === 'object' && !Array.isArray(r.fields)) {
   443→    const src = r.fields as Record<string, unknown>
   444→    const fields: PageFields = {}
   445→    let n = 0
   446→    for (const [k, fr] of Object.entries(src)) {
   447→      if (n >= 16) break // 字段数上限(正常规则 ≤8 字段)
   448→      const sanitized = sanitizeFieldRule(fr)
   449→      if (sanitized) { fields[k.slice(0, 40)] = sanitized; n++ }
   450→    }
   451→    out.fields = fields
   452→  }
   453→  if (r.pagination && typeof r.pagination === 'object' && !Array.isArray(r.pagination)) {
   454→    const p = r.pagination as Record<string, unknown>
   455→    const nextLink = sanitizeFieldRule(p.nextLink)
   456→    out.pagination = {
   457→      enabled: safeBool(p.enabled) ?? false,
   458→      maxPages: safeNum(p.maxPages, 1, 500) ?? 20,
   459→      joinWith: safeStr(p.joinWith, 200) ?? '',
   460→      ...(nextLink ? { nextLink } : {}),
   461→    }
   462→  }
   463→  return out
   464→}

   466→/** FetchConfig 消毒: 引擎/UA模式白名单 + 数值钳制(字符串数字转真数字) */
   467→export function sanitizeFetchConfig(v: unknown): Partial<FetchConfig> {
   468→  const out: Partial<FetchConfig> = {}
   469→  if (!v || typeof v !== 'object' || Array.isArray(v)) return out
   470→  const r = v as Record<string, unknown>
   471→  if (r.engine === 'auto' || r.engine === 'http' || r.engine === 'browser') out.engine = r.engine
   472→  if (
   473→    r.uaMode === 'rotate' || r.uaMode === 'fixed' || r.uaMode === 'custom' ||
   474→    r.uaMode === 'mobile' || r.uaMode === 'desktop'
   475→  ) out.uaMode = r.uaMode
   476→  const customUaRaw = safeStr(r.customUa, 300)
   477→  if (customUaRaw !== undefined) {
   478→    const customUa = safeSingleLine(customUaRaw)
   479→    if (customUa) out.customUa = customUa
   480→  }
   481→  if (r.headers && typeof r.headers === 'object' && !Array.isArray(r.headers)) {
   482→    const src = r.headers as Record<string, unknown>
   483→    const headers: Record<string, string> = {}
   484→    let n = 0
   485→    for (const [k, val] of Object.entries(src)) {
   486→      if (n >= 30) break
   487→      const key = safeHeaderKey(k.slice(0, 100))
   488→      const s = safeStr(val, 1000)
   489→      if (key && s !== undefined) {
   490→        const value = safeSingleLine(s)
   491→        if (value) { headers[key] = value; n++ }
   492→      }
   493→    }
   494→    if (n > 0) out.headers = headers
   495→  }
   496→  const cookiesRaw = safeStr(r.cookies, 4000)
   497→  if (cookiesRaw !== undefined) {
   498→    const cookies = safeSingleLine(cookiesRaw)
   499→    if (cookies) out.cookies = cookies
   500→  }
   501→  const autoCookie = safeBool(r.autoCookie)
   502→  if (autoCookie !== undefined) out.autoCookie = autoCookie
   503→  const referer = safeBool(r.referer)
   504→  if (referer !== undefined) out.referer = referer
   505→  // Referer 链伪造(ff-b): 规则级开关可序列化入库; refererUrl 为运行时逐请求注入项
   506→  // (同 pageFetch 口径)刻意不进白名单
   507→  const refererChain = safeBool(r.refererChain)
   508→  if (refererChain !== undefined) out.refererChain = refererChain
   509→  const timeout = safeNum(r.timeout, 1000, 120_000)
   510→  if (timeout !== undefined) out.timeout = timeout
   511→  const retries = safeNum(r.retries, 0, 5)
   512→  if (retries !== undefined) out.retries = retries
   513→  const waitSelector = safeStr(r.waitSelector, 300)
   514→  if (waitSelector !== undefined) out.waitSelector = waitSelector
   515→  const waitMs = safeNum(r.waitMs, 0, 60_000)
   516→  if (waitMs !== undefined) out.waitMs = waitMs
   517→  const clickSelector = safeStr(r.clickSelector, 300)
   518→  if (clickSelector !== undefined) out.clickSelector = clickSelector
   519→  if (Array.isArray(r.browserFallbackStatus)) {
   520→    const statuses = r.browserFallbackStatus
   521→      .map((x) => safeNum(x, 100, 599))
   522→      .filter((x): x is number => x !== undefined)
   523→      .slice(0, 10)
   524→    if (statuses.length) out.browserFallbackStatus = statuses
   525→  }
   526→  // hostGate 同站并发闸门上限: 1~10 钳制(与 hostgate.clampLimit 口径一致),
   527→  // parseRuleConfig 经 base.fetch 合并缺省 3
   528→  const hostGateLimit = safeNum(r.hostGateLimit, 1, 10)
   529→  if (hostGateLimit !== undefined) out.hostGateLimit = hostGateLimit
   530→  // 通用 token 预取钩子(bb-d): 白名单同步(字符串钳长/枚举白名单), 非法类型丢弃。
   531→  // pageFetch 为运行时注入项刻意不进白名单(函数不可 JSON 序列化, 规则 JSON 同名键自然丢弃)
   532→  const tokenUrl = safeStr(r.tokenUrl, 1000)
   533→  if (tokenUrl !== undefined) out.tokenUrl = tokenUrl
   534→  const tokenPattern = safeStr(r.tokenPattern, 300)
   535→  if (tokenPattern !== undefined) out.tokenPattern = tokenPattern
   536→  if (r.tokenInjection === 'url' || r.tokenInjection === 'header') out.tokenInjection = r.tokenInjection
   537→  const tokenHeaderName = safeStr(r.tokenHeaderName, 100)
   538→  if (tokenHeaderName !== undefined) out.tokenHeaderName = tokenHeaderName
   539→  // 内容代理 URL(feat-contentproxy-resume): 钳长 500(代理 URL 含查询参数足够) +
   540→  // 单行化(防 CR/LF 注入), 非法形态整字段丢弃。运行时仅 http(s) 形态经 assertSafeTarget
   541→  // 放行(loopback 允许, 与 tokenUrl 同口径), 含 {url} 占位符由 fetcher.split('{url}')
   542→  // .join(enc) 全量替换(prefetchToken 同款实现)
   543→  const contentProxyUrlRaw = safeStr(r.contentProxyUrl, 500)
   544→  if (contentProxyUrlRaw !== undefined) {
   545→    const contentProxyUrl = safeSingleLine(contentProxyUrlRaw)
   546→    if (contentProxyUrl && /^https?:\/\/\S+$/i.test(contentProxyUrl)) out.contentProxyUrl = contentProxyUrl
   547→  }
   548→  // 出口代理池(dd-a): 钳长 2000(多条列表形态) + 逐条 scheme 白名单校验
   549→  // (http/https/socks5(h)/socks4(a) + host:port 形态, 与 fetcher.parseProxyPool 同口径);
   550→  // 合法条目去空去重上限 10 条后回写, 全部非法则整字段丢弃
   551→  const proxyUrl = safeStr(r.proxyUrl, 2000)
   552→  if (proxyUrl !== undefined) {
   553→    const valid: string[] = []
   554→    const seen = new Set<string>()
   555→    for (const raw of proxyUrl.split(',')) {
   556→      const s = raw.trim()
   557→      if (!s || seen.has(s) || !isValidProxySpec(s) || valid.length >= 10) continue
   558→      seen.add(s)
   559→      valid.push(s)
   560→    }
   561→    if (valid.length) out.proxyUrl = valid.join(',')
   562→  }
   563→  // 镜像域名组(dd-b): 钳长 2000(多条列表形态) + 逐条域名校验(host[:port] 形态, 不带
   564→  // scheme/path, 单条 ≤500) + 去空去重(大小写不敏感, 统一小写)上限 10 条后回写,
   565→  // 全部非法则整字段丢弃 —— 口径对齐 proxyUrl 惯例
   566→  const mirrorDomains = safeStr(r.mirrorDomains, 2000)
   567→  if (mirrorDomains !== undefined) {
   568→    const valid: string[] = []
   569→    const seen = new Set<string>()
   570→    for (const raw of mirrorDomains.split(',')) {
   571→      const s = raw.trim().toLowerCase()
   572→      if (!s || seen.has(s) || !isValidMirrorHost(s) || valid.length >= 10) continue
   573→      seen.add(s)
   574→      valid.push(s)
   575→    }
   576→    if (valid.length) out.mirrorDomains = valid.join(',')
   577→  }
   578→  // feat-round-8: Feature B3 — 代理轮换策略白名单
   579→  if (
   580→    r.proxyRotationStrategy === 'round-robin' ||
   581→    r.proxyRotationStrategy === 'random' ||
   582→    r.proxyRotationStrategy === 'least-used'
   583→  ) out.proxyRotationStrategy = r.proxyRotationStrategy
   584→  // feat-round-8: Feature B1 — 请求抖动 0~30000ms 钳制(超过 30s 抖动已是离谱配置,
   585→  // 上限防止误填 60000 当分钟值跑; 钳到 30s 仍允许极端慢站手工配)
   586→  const jitterMs = safeNum(r.jitterMs, 0, 30_000)
   587→  if (jitterMs !== undefined) out.jitterMs = jitterMs
   588→  // 采集传输模式(hh-c): 枚举白名单校验 —— 非法值丢弃(undefined = 回退 native 链零回归);
   589→  // 'native' 显式接受(语义等价缺省)。scraplingBridgeUrl: safeStr 钳长 + http(s) URL 形态
   590→  // 校验 + 单行化(防 CR/LF 注入), 非法形态整字段丢弃
   591→  if (
   592→    r.fetchMode === 'native' || r.fetchMode === 'scrapling-static' ||
   593→    r.fetchMode === 'scrapling-stealthy' || r.fetchMode === 'scrapling-playwright'
   594→  ) out.fetchMode = r.fetchMode
   595→  const scraplingBridgeUrlRaw = safeStr(r.scraplingBridgeUrl, 300)
   596→  if (scraplingBridgeUrlRaw !== undefined) {
   597→    const scraplingBridgeUrl = safeSingleLine(scraplingBridgeUrlRaw)
   598→    if (scraplingBridgeUrl && /^https?:\/\/\S+$/i.test(scraplingBridgeUrl)) out.scraplingBridgeUrl = scraplingBridgeUrl
   599→  }
   600→  return out
   601→}

   603→/** 镜像条目形态校验(types.ts 消毒与 fetcher.ts 运行时组解析共用口径, 改动需同步):
   604→ *  无 scheme/无 path/无空白的 host[:port] 形态(host 标签字母数字连字符, 允许纯数字 IP
   605→ *  如 127.0.0.1 与 localhost), 单条 ≤500 字符; 不支持 IPv6 字面量(冒号与端口定界冲突, 列表形态固有) */
   606→export function isValidMirrorHost(s: string): boolean {
   607→  return s.length <= 500 && /^[a-z0-9]([a-z0-9-]*[a-z0-9])?(\.[a-z0-9]([a-z0-9-]*[a-z0-9])?)*(:\d{1,5})?$/.test(s)
   608→}

   619→// 是入口防线; 引擎侧 try/catch 静默忽略语义保留(运行时韧性兜底, 双层防线)。
   620→// 零误伤要求: 规则库全部存量规则的正则字段必须全数通过(verify-gg-a-regex 实证)
   621→// ============================================================

   623→export interface RegexSafetyResult {
   624→  ok: boolean
   625→  reason?: string
   626→}

   628→/** 无界量词(+ 号、星号、{n,} 开区间)识别: 返回量词长度(含懒惰标记 ?), 非量词返回 0 */
   629→function unboundedQuantifierLen(p: string, i: number): number {
   630→  const c = p[i]
   631→  if (c === '+' || c === '*') return p[i + 1] === '?' ? 2 : 1
   632→  if (c === '{') {
   633→    const m = /^\{(\d+),(\d*)\}/.exec(p.slice(i))
   634→    if (m && m[2] === '') return m[0].length + (p[i + m[0].length] === '?' ? 1 : 0)
   635→  }
   636→  return 0
   637→}

   639→/** 跳过一个正则原子(转义序列/字符类/嵌套组/单字符), 返回原子结束下标(不含其量词) */
   640→function skipRegexAtom(p: string, i: number, end: number): number {
   641→  const c = p[i]
   642→  if (c === '\\') return Math.min(end, i + 2)
   643→  if (c === '[') {
   644→    let j = i + 1
   645→    if (p[j] === '^') j++
   646→    if (p[j] === ']') j++ // 首位 ] 为字面量
   647→    while (j < end && p[j] !== ']') {
   648→      if (p[j] === '\\') j++
   649→      j++
   650→    }
   651→    return Math.min(end, j + 1)
   652→  }
   653→  if (c === '(') {
   654→    let depth = 1
   655→    let j = i + 1
   656→    while (j < end && depth > 0) {
   657→      const cj = p[j]
   658→      if (cj === '\\') { j += 2; continue }
   659→      if (cj === '[') {
   660→        let k = j + 1
   661→        if (p[k] === '^') k++
   662→        if (p[k] === ']') k++
   663→        while (k < end && p[k] !== ']') {
   664→          if (p[k] === '\\') k++
   665→          k++
   666→        }
   667→        j = k + 1
   668→        continue
   669→      }
   670→      if (cj === '(') depth++
   671→      else if (cj === ')') depth--
   672→      j++
   673→    }
   674→    return Math.min(end, j)
   675→  }
   676→  return Math.min(end, i + 1)
   677→}
   678→
   679→/**
   680→ * 组内容(from..to)按顶层 | 分支检查: 任一分支的末原子带无界量词 → true。
   681→ * 覆盖 (a+)+、(X*)×星、(a{2,})+、(?:\w+)+ 等显式嵌套, 以及 (a+|b)+ 型"分支歧义"形态
   682→ * (末分支虽无嵌套, 但量词分支可变长 + 外层无界循环 = 同型灾难回溯)。
   683→ * 保守口径: 有界外层({n,m})/末原子有界(\d+\.)+ 均不报警, 避免误伤正常模式
   684→ */
   685→function groupHasUnboundedTail(p: string, from: number, to: number): boolean {
   686→  let i = from
   687→  let branchUnbounded = false
   688→  let anyBranchUnbounded = false
   689→  while (i < to) {
   690→    if (p[i] === '|') {
   691→      anyBranchUnbounded = anyBranchUnbounded || branchUnbounded
   692→      branchUnbounded = false
   693→      i++
   694→      continue
   695→    }
   696→    const atomEnd = skipRegexAtom(p, i, to)
   697→    const q = unboundedQuantifierLen(p, atomEnd)
   698→    branchUnbounded = q > 0
   699→    i = atomEnd + q
   700→  }
   701→  return anyBranchUnbounded || branchUnbounded
   702→}

   704→/** 顶层 | 分支的字面量串提取(rr-c3): 分支内仅含普通字符与单字符转义(无量词/类/组/元字符)
   705→ *  时返回解码后的字面量串; 含任何元字符(量词/类/组/锚/点/类缩写)返回 null(保守不参与判定) */
   706→function literalBranchOf(p: string, from: number, to: number): string | null {
   707→  let out = ''
   708→  let i = from
   709→  while (i < to) {
   710→    const c = p[i]
   711→    if (c === '\\') {
   712→      if (i + 1 >= to) return null
   713→      const n = p[i + 1]
   714→      if ('dDwWsSbB'.includes(n)) return null // 类缩写非字面量
   715→      out += n === 'n' ? '\n' : n === 't' ? '\t' : n === 'r' ? '\r' : n
   716→      i += 2
   717→      continue
   718→    }
   719→    if ('+*?[](){}|^$.'.includes(c)) return null
   720→    out += c
   721→    i++
   722→  }
   723→  return out
   724→}

   726→/** 分支前缀歧义(rr-c3): 组带无界量词时, 若存在两个字面量分支互为前缀(如 a 与 aa),
   727→ *  匹配路径数随输入长度指数增长 —— (a|aa)+ 是 OWASP 经典灾难形态, 与 (a+)+ 同级;
   728→ *  生产引擎运行时 node/V8 无 JSC 式回溯预算(probe-rr-c3-redos 实测: 41 字符输入 47s,
   729→ *  53 字符 15s 跑不完), 原校验器仅认显式嵌套量词漏过此类。分支拆分复用 skipRegexAtom
   730→ *  (嵌套组/类整体消耗, '|' 必为组内顶层), 字面量分支互为前缀即报警(保守口径) */
   731→function branchesHavePrefixAmbiguity(p: string, from: number, to: number): boolean {
   732→  const literals: (string | null)[] = []
   733→  let i = from
   734→  let start = from
   735→  while (i < to) {
   736→    const atomEnd = skipRegexAtom(p, i, to)
   737→    i = atomEnd + unboundedQuantifierLen(p, atomEnd)
   738→    if (i < to && p[i] === '|') {
   739→      literals.push(literalBranchOf(p, start, i))
   740→      i++
   741→      start = i
   742→    }
   743→  }
   744→  literals.push(literalBranchOf(p, start, to))
   745→  const lits = literals.filter((s): s is string => s !== null && s.length > 0)
   746→  for (let a = 0; a < lits.length; a++) {
   747→    for (let b = 0; b < lits.length; b++) {
   748→      if (a !== b && lits[b].startsWith(lits[a])) return true
   749→    }
   750→  }
   751→  return false
   752→}
   753→
   754→/** 嵌套量词检测(保守启发式):
   755→ *  ① 结构检查: 未转义组 (...) 后跟无界量词, 且组内任一分支以无界量词原子收尾;
   756→ *  ② 分支歧义检查(rr-c3 bug#4): 组带无界量词且组内两个字面量分支互为前缀((a|aa)+ 型,
   757→ *     OWASP 经典灾难形态, 与显式嵌套同级 —— 生产引擎运行时 node/V8 无 JSC 式回溯预算,
   758→ *     probe-rr-c3-redos 实测 41 字符输入 47s、53 字符 15s 跑不完);
   759→ *  ③ 形态检查: [+*])[+*{ 与 cleaner.removeAdLines 运行时跳过闸门同款
   760→ *     (该闸门会静默跳过此形态的模式 —— API 层若放行会造成"校验通过但运行时静默失效") */
   761→export function hasNestedQuantifier(pattern: string): boolean {
   762→  const n = pattern.length
   763→  let i = 0
   764→  const groupStack: number[] = []
   765→  while (i < n) {
   766→    const c = pattern[i]
   767→    if (c === '\\') { i += 2; continue }
   768→    if (c === '[') {
   769→      let j = i + 1
   770→      if (pattern[j] === '^') j++
   771→      if (pattern[j] === ']') j++
   772→      while (j < n && pattern[j] !== ']') {
   773→        if (pattern[j] === '\\') j++
   774→        j++
   775→      }
   776→      i = j + 1
   777→      continue
   778→    }
   779→    if (c === '(') { groupStack.push(i); i++; continue }
   780→    if (c === ')') {
   781→      const start = groupStack.pop()
   782→      i++
   783→      const q = unboundedQuantifierLen(pattern, i)
   784→      if (q > 0 && start !== undefined) {
   785→        if (groupHasUnboundedTail(pattern, start + 1, i - 1)) return true
   786→        if (branchesHavePrefixAmbiguity(pattern, start + 1, i - 1)) return true
   787→      }
   788→      i += q
   789→      continue
   790→    }
   791→    i++
   792→  }
   793→  // 形态检查(与 cleaner 运行时闸门同口径): 量词紧跟闭括号再跟量词/开花括号
   794→  if (/[+*]\s*\)\s*[+*{]/.test(pattern)) return true
   795→  return false
   796→}

   800→ * ②嵌套量词启发式(灾难回溯 → 拒绝)。flags 传引擎实际生效的标志位
