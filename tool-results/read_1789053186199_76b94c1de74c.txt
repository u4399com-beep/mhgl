    67→// 卡片阴影映射(由配色 primary 派生, 透传给 generateTheme 在合成期生成实际值)
    68→function shadowOf(kind: ShadowKind, primary: string, dark: boolean): string {
    69→  switch (kind) {
    70→    case 'none': return 'none'
    71→    case 'sm': return dark ? `0 2px 6px rgba(0,0,0,0.4)` : `0 2px 6px rgba(0,0,0,0.08)`
    72→    case 'md': return dark ? `0 6px 18px rgba(0,0,0,0.5)` : `0 6px 18px rgba(0,0,0,0.12)`
    73→    case 'lg': return dark ? `0 12px 36px rgba(0,0,0,0.6)` : `0 12px 36px rgba(0,0,0,0.16)`
    74→    case 'glow': {
    75→      // primary 色辉光
    76→      const m = /^#([0-9a-f]{6})$/i.exec(primary.trim())
    77→      if (m) {
    78→        const n = parseInt(m[1], 16)
    79→        const r = (n >> 16) & 255, g = (n >> 8) & 255, b = n & 255
    80→        return `0 8px 32px rgba(${r},${g},${b},0.35)`
    81→      }
    82→      return `0 8px 32px ${primary}`
    83→    }
    84→    case 'depth': return dark ? `0 20px 60px rgba(0,0,0,0.7), 0 8px 16px rgba(0,0,0,0.5)` : `0 20px 60px rgba(0,0,0,0.18), 0 8px 16px rgba(0,0,0,0.08)`
    85→  }
    86→}

   237→// ============================================================
   238→// 4. 合成函数: colorId + styleId + layoutId → ThemeDef
   239→// ============================================================
   240→export function generateTheme(colorSchemeId: string, styleId: string, layoutId: string): ThemeDef | undefined {
   241→  const c = COLOR_BY_ID.get(colorSchemeId)
   242→  const s = STYLE_BY_ID.get(styleId)
   243→  const l = LAYOUT_BY_ID.get(layoutId)
   244→  if (!c || !s || !l) return undefined
   245→  const id = `${c.id}-${s.id}-${l.id}`
   246→  // 名称: 配色名 + 风格名 + 布局名
   247→  const name = `${c.name}·${s.name}·${l.name}`
   248→  // 描述: 风格 desc + 配色 dark/light + 布局 read kind
   249→  const desc = `${s.desc} · ${c.dark ? '暗色' : '亮色'} · ${l.name}`
   250→  // read: 布局自带 readVars, 但 texture/chapterDeco 由 style 覆盖(保留视觉一致性)
   251→  const read: ThemeReadConfig = {
   252→    ...l.readVars,
   253→    texture: s.texture,
   254→    chapterDeco: s.chapterDeco,
   255→  }
   256→  return {
   257→    id,
   258→    name,
   259→    desc,
   260→    layout: l.homeLayout,
   261→    dark: c.dark,
   262→    read,
   263→    vars: {
   264→      bg: c.bg,
   265→      surface: c.surface,
   266→      surfaceAlt: c.surfaceAlt,
   267→      text: c.text,
   268→      textMuted: c.textMuted,
   269→      primary: c.primary,
   270→      primaryText: c.primaryText,
   271→      accent: c.accent,
   272→      border: c.border,
   273→      radius: `${s.radius}px`,
   274→      fontFamily: FONT_FAMILIES[s.fontFamily],
   275→      cardShadow: shadowOf(s.cardShadow, c.primary, c.dark),
   276→      headerStyle: s.headerStyle,
   277→      titleFont: s.fontFamily === 'serif' ? '"Noto Serif SC","Songti SC",serif' : undefined,
   278→    },
   279→    preview: c.preview,
   280→  }
   281→}

   283→/** 解析主题 ID → (colorId, styleId, layoutId)
   284→ *  ID 形如 `violet-glasswa-grid-cl` —— 但 colorId 自身可含 `-`(如 `vio-gold-d`),
   285→ *  故按反向拆分: 末段为 layoutId(24 选 1), 倒数第二段为 styleId(42 选 1), 其余为 colorId。
   286→ *  通过对预定义 ID 列表 set 匹配消歧。 */
   287→export function parseThemeId(themeId: string): { colorId: string; styleId: string; layoutId: string } | undefined {
   288→  if (!themeId || typeof themeId !== 'string') return undefined
   289→  // 优先精确匹配预定义 layout/style/color id 组合
   290→  // 先尝试匹配 layout (24 个, ID 唯一)
   291→  for (const l of LAYOUTS) {
   292→    if (themeId.endsWith(`-${l.id}`)) {
   293→      const rest = themeId.slice(0, themeId.length - l.id.length - 1) // 去掉 `-${l.id}`
   294→      // 再匹配 style (42 个, ID 唯一)
   295→      for (const s of STYLES) {
   296→        if (rest.endsWith(`-${s.id}`)) {
   297→          const colorId = rest.slice(0, rest.length - s.id.length - 1)
   298→          if (COLOR_BY_ID.has(colorId)) {
   299→            return { colorId, styleId: s.id, layoutId: l.id }
   300→          }
   301→        }
   302→      }
   303→    }
   304→  }
   305→  return undefined
   306→}
   307→
   308→/** 单一解析: 主题 ID → 完整 ThemeDef */
   309→export function getThemeById(themeId: string): ThemeDef | undefined {
   310→  if (!themeId) return undefined
   311→  const parsed = parseThemeId(themeId)
   312→  if (!parsed) return undefined
   313→  return generateTheme(parsed.colorId, parsed.styleId, parsed.layoutId)
   314→}

   319→export interface ThemeListItem {
   320→  id: string
   321→  name: string
   322→  desc: string
   323→  layout: HomeLayoutKind
   324→  dark: boolean
   325→  read?: { layout: ReadLayoutKind }
   326→  preview: [string, string, string]
   327→}

   329→/** 全部 50400 组合的轻量描述符列表(惰性生成, 单次访问)
   330→ *  注意: 全量生成约 50400 项 × ~200B = ~10MB 内存占用, 仅 admin 列表 API 单次构建后即丢;
   331→ *  长期持有的 admin/themes 路由每次请求 lazy 重新构建, 不长期驻留 */
   332→export function getThemeList(): ThemeListItem[] {
   333→  const out: ThemeListItem[] = []
   334→  for (const c of COLOR_SCHEMES) {
   335→    for (const s of STYLES) {
   336→      for (const l of LAYOUTS) {
   337→        out.push({
   338→          id: `${c.id}-${s.id}-${l.id}`,
   339→          name: `${c.name}·${s.name}·${l.name}`,
   340→          desc: `${s.desc} · ${c.dark ? '暗色' : '亮色'} · ${l.name}`,
   341→          layout: l.homeLayout,
   342→          dark: c.dark,
   343→          read: { layout: l.readLayout },
   344→          preview: c.preview,
   345→        })
   346→      }
   347→    }
   348→  }
   349→  return out
   350→}

   352→/** 分页返回主题列表(1-based, 1..N) */
   353→export function getThemesPage(page: number, size: number): {
   354→  page: number
   355→  size: number
   356→  total: number
   357→  totalPages: number
   358→  items: ThemeListItem[]
   359→} {
   360→  const all = getThemeList()
   361→  const total = all.length
   362→  const totalPages = Math.max(1, Math.ceil(total / Math.max(1, size)))
   363→  const p = Math.max(1, Math.min(totalPages, Math.floor(page) || 1))
   364→  const sz = Math.max(1, Math.min(500, Math.floor(size) || 50))
   365→  const start = (p - 1) * sz
   366→  const items = all.slice(start, start + sz)
   367→  return { page: p, size: sz, total, totalPages, items }
   368→}
