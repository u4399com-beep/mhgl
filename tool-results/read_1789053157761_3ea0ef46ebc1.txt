    67→/** 取/创建会话种子: 同 URL host 复用种子, 维持会话内指纹一致性 */
    68→function getSeed(sessionKey: string): SessionSeed {
    69→  const existing = sessions.get(sessionKey)
    70→  // 30 分钟过期: 同 host 长任务下定期轮换种子, 防长期指纹跟踪
    71→  if (existing && Date.now() - existing.createdAt < 30 * 60 * 1000) return existing
    72→  // 用 crypto 生成 15 位 IMEI-like device ID(用于 navigator.userAgentData + cookie)
    73→  const buf = new Uint8Array(8)
    74→  for (let i = 0; i < 8; i++) buf[i] = Math.floor(Math.random() * 256)
    75→  // IMEI 格式: 15 位数字(0-9), 前缀 86(中国区段)
    76→  let deviceId = '86'
    77→  for (let i = 0; i < 13; i++) deviceId += String(buf[i % buf.length] % 10)
    78→  const seed: SessionSeed = {
    79→    canvasNoise: Math.floor(Math.random() * 1_000_000_000),
    80→    audioNoise: Math.floor(Math.random() * 1_000_000_000),
    81→    deviceId,
    82→    createdAt: Date.now(),
    83→  }
    84→  // LRU 防泄漏: 上限 200 会话
    85→  if (sessions.size > 200) {
    86→    const firstKey = sessions.keys().next().value
    87→    if (firstKey) sessions.delete(firstKey)
    88→  }
    89→  sessions.set(sessionKey, seed)
    90→  return seed
    91→}

   487→async function enableRequestInterception(page: any): Promise<void> {
   488→  try {
   489→    await page.setRequestInterception(true)

   520→async function tryClickTurnstile(page: any): Promise<void> {
   521→  try {
   522→    const frames = page.frames()
   523→    for (let i = 0; i < frames.length && i < 8; i++) {
   524→      try {
   525→        const frameUrl = frames[i].url() || ''
   526→        const isCfFrame = /challenges\.cloudflare\.com|cdn-cgi\/challenge-platform/.test(frameUrl)
   527→        const sel = isCfFrame ? 'input[type=checkbox]' : '.cf-turnstile input[type=checkbox]'
   528→        const cnt = await frames[i].locator(sel).count().catch(() => 0)
   529→        if (cnt === 0) continue
   530→        await frames[i].click(sel, { timeout: 1500 }).catch(() => {})
   531→        return
   532→      } catch { /* 静默: 换下一个 frame */ }
   533→    }
   534→  } catch { /* 整体容忍 */ }
   535→}

   537→/** 等 CF challenge 自动放行, 最长 30s, 期间每 2s 尝试点 Turnstile checkbox */
   538→async function waitCfChallenge(page: any, maxMs = 30000): Promise<boolean> {
   539→  const start = Date.now()
   540→  while (Date.now() - start < maxMs) {
   541→    await new Promise((r) => setTimeout(r, 2000))
   542→    await tryClickTurnstile(page)
   543→    try {
   544→      const html = await page.content()
   545→      if (!isCfChallenge(html)) return true
   546→    } catch { /* 页面 reload 中, 继续 */ }
   547→  }
   548→  return false
   549→}

   551→/** 兜底: 先访问 /robots.txt(轻量页面) 拿 cf_clearance, 再回访目标 */
   552→async function tryCfClearanceFallback(page: any, targetUrl: string, timeoutMs: number): Promise<boolean> {
   553→  try {
   554→    const u = new URL(targetUrl)
   555→    const robotsUrl = `${u.origin}/robots.txt`
   556→    await page.goto(robotsUrl, { waitUntil: 'domcontentloaded', timeout: Math.min(timeoutMs, 15000) }).catch(() => {})
   557→    await new Promise((r) => setTimeout(r, 1500))
   558→    // 再回访目标
   559→    const resp = await page.goto(targetUrl, { waitUntil: 'networkidle2', timeout: timeoutMs }).catch(() => null)
   560→    if (resp) {
   561→      const html = await page.content().catch(() => '')
   562→      return !isCfChallenge(html) && html.length > 200
   563→    }
   564→  } catch { /* 兜底失败容忍 */ }
   565→  return false
   566→}

   570→async function fetchPage(url: string, tier: StealthTier, timeoutMs: number): Promise<FetchResult> {
   571→  const b = await ensureBrowser()
   572→  const page = await b.newPage()
   573→  await page.setViewport({ width: 1920, height: 1080 })
   574→  await page.setUserAgent(DEFAULT_UA)
   575→
   576→  const sessionKey = sessionKeyOf(url)
   577→  const seed = getSeed(sessionKey)
   578→
   579→  // 标准档+: CDP UA override + stealth 脚本注入
   580→  if (tier === 'standard' || tier === 'maximum') {
   581→    await applyCdpUaOverride(page, DEFAULT_UA)
   582→    const scripts = [buildStandardStealthScript(seed)]
   583→    if (tier === 'maximum') scripts.push(buildMaximumStealthScript(seed))
   584→    await injectStealthScripts(page, scripts)
   585→    await applyDeviceMetrics(page, 1920, 1080)
   586→  }
   587→
   588→  // 最大档位: request interception (屏蔽 tracking/ads)
   589→  if (tier === 'maximum') {
   590→    await enableRequestInterception(page)
   591→  }
   592→
   593→  let status = 0
   594→  let finalUrl = url
   595→
   596→  try {
   597→    const response = await page.goto(url, { waitUntil: 'networkidle2', timeout: timeoutMs })
   598→    if (response) {
   599→      status = response.status()
   600→      finalUrl = page.url()
   601→    }
   602→
   603→    let html = await page.content()
   604→
   605→    // CF challenge 检测: 等 30s + 尝试点 Turnstile
   606→    if (isCfChallenge(html)) {
   607→      const cleared = await waitCfChallenge(page, 30000)
   608→      if (!cleared) {
   609→        // 兜底: 先访问 /robots.txt 拿 cf_clearance
   610→        const ok = await tryCfClearanceFallback(page, url, timeoutMs)
   611→        if (ok) {
   612→          finalUrl = page.url()
   613→          status = 200
   614→        }
   615→      }
   616→      await new Promise((r) => setTimeout(r, 1500))
   617→      html = await page.content().catch(() => html)
   618→    }
   619→
   620→    const cookies = await page.cookies()
   621→    return { ok: html.length > 100, html, status, finalUrl, cookies: cookies as unknown as Array<Record<string, unknown>>, tier }
   622→  } catch (e: any) {
   623→    const html = await page.content().catch(() => '')
   624→    return { ok: false, html, status, finalUrl, cookies: [], tier }
   625→  } finally {
   626→    await page.close().catch(() => {})
   627→  }
   628→}

   694→// feat-cloak-anticrawler E: SIGTERM 优雅关闭 —— 持久浏览器实例 + 在飞请求计数等待,
   695→// 让父进程(Next.js)发 SIGTERM 时本服务也能干净退出, 不留 orphan chromium 进程。

   697→let cloakShutdownInProgress = false
   698→async function shutdownCloakBrowser(sig: string): Promise<void> {
   699→  if (cloakShutdownInProgress) return

   703→  const waitDeadline = Date.now() + 10_000
   704→  while (inFlight > 0 && Date.now() < waitDeadline) {
   705→    await new Promise((r) => setTimeout(r, 200))
   706→  }
   707→  if (inFlight > 0) {
   708→    console.warn(`[cloak-browser] 优雅关闭超时, 仍有 ${inFlight} 个在飞请求, 强制退出`)
   709→  }
   710→  try {
   711→    if (browser) {
   712→      await browser.close().catch(() => {})
   713→      browser = null
   714→    }
   715→  } catch { /* 已死则忽略 */ }
   716→  process.exit(0)
   717→}
